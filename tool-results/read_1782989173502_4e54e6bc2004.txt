    82→    82→/** % change of second half of series vs first half. */
    83→    83→function computeDelta(series: AnalyticsPoint[], key: MetricKey): number | undefined {
    84→    84→  if (series.length < 4) return undefined;
    85→    85→  const mid = Math.floor(series.length / 2);
    86→    86→  const first = series.slice(0, mid).reduce((s, p) => s + p[key], 0);
    87→    87→  const second = series.slice(mid).reduce((s, p) => s + p[key], 0);
    88→    88→  if (first === 0) return undefined;
    89→    89→  return Math.round(((second - first) / first) * 100);
    90→    90→}

   555→   555→function Funnel({ totals }: { totals: AnalyticsTotals }) {
   556→   556→  const steps = [
   557→   557→    { label: "Search Views", value: totals.searchViews, key: "searchViews" as MetricKey, icon: Search },
   558→   558→    { label: "Maps Views", value: totals.mapsViews, key: "mapsViews" as MetricKey, icon: Map },
   559→   559→    { label: "Website Clicks", value: totals.websiteClicks, key: "websiteClicks" as MetricKey, icon: MousePointerClick },
   560→   560→    { label: "Phone Calls", value: totals.phoneCalls, key: "phoneCalls" as MetricKey, icon: Phone },
   561→   561→    { label: "Directions", value: totals.directionRequests, key: "directionRequests" as MetricKey, icon: Navigation },
   562→   562→  ];
   563→   563→  const max = steps[0]?.value || 1;
   564→   564→  return (
   565→   565→    <div className="space-y-2.5">
   566→   566→      {steps.map((s, i) => {
   567→   567→        const widthPct = Math.max(2, Math.round((s.value / max) * 100));
   568→   568→        const convPct = i === 0 ? 100 : Math.round((s.value / (steps[i - 1].value || 1)) * 100);
   569→   569→        return (
   570→   570→          <div key={s.key}>
   571→   571→            <div className="flex items-center justify-between text-xs mb-1">
   572→   572→              <span className="font-medium flex items-center gap-1.5">
   573→   573→                <s.icon className="size-3.5" style={{ color: METRIC_COLORS[s.key] }} />
   574→   574→                {s.label}
   575→   575→              </span>
   576→   576→              <span className="tabular-nums text-muted-foreground">
   577→   577→                <span className="text-foreground font-semibold">{fmt(s.value)}</span>
   578→   578→                {i > 0 && <span className="ml-1.5 text-[10px]">({convPct}%)</span>}
   579→   579→              </span>
   580→   580→            </div>
   581→   581→            <div className="h-6 rounded-md bg-muted overflow-hidden">
   582→   582→              <div

   584→   584→                style={{ width: `${widthPct}%`, backgroundColor: METRIC_HEX[s.key] }}

   591→   591→      })}
   592→   592→      <div className="pt-2 mt-2 border-t text-[11px] text-muted-foreground leading-relaxed">

   595→   595→        </span> of discovery actions converted to engagement.
