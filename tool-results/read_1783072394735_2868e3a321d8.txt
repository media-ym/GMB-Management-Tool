   165→// ── Rank → color helpers ─────────────────────────────────────────────
   166→function rankColor(rank: number): { bg: string; text: string; label: string } {
   167→  if (!rank || rank <= 0) return { bg: "bg-slate-300 dark:bg-slate-700", text: "text-slate-600 dark:text-slate-300", label: "Not ranked" };
   168→  if (rank <= 3) return { bg: "bg-emerald-500", text: "text-white", label: "Top 3" };
   169→  if (rank <= 10) return { bg: "bg-amber-500", text: "text-white", label: "Top 10" };
   170→  if (rank <= 20) return { bg: "bg-orange-500", text: "text-white", label: "Top 20" };
   171→  return { bg: "bg-rose-500", text: "text-white", label: "21+" };
   172→}
   173→
   174→function rankTextClass(rank: number): string {
   175→  if (!rank || rank <= 0) return "text-slate-500";
   176→  if (rank <= 3) return "text-emerald-600 dark:text-emerald-400";
   177→  if (rank <= 10) return "text-amber-600 dark:text-amber-400";
   178→  if (rank <= 20) return "text-orange-600 dark:text-orange-400";
   179→  return "text-rose-600 dark:text-rose-400";
   180→}
   181→
   182→function scoreBg(score: number): string {
   183→  if (score >= 75) return "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400";
   184→  if (score >= 50) return "bg-amber-500/15 text-amber-700 dark:text-amber-400";
   185→  return "bg-rose-500/15 text-rose-700 dark:text-rose-400";
   186→}
   187→
   188→function rankBandClass(rank: number): string {
   189→  if (!rank || rank <= 0) return "bg-slate-500/15 text-slate-600 dark:text-slate-400";
   190→  if (rank <= 3) return "bg-emerald-500/15 text-emerald-700 dark:text-emerald-400";
   191→  if (rank <= 10) return "bg-amber-500/15 text-amber-700 dark:text-amber-400";
   192→  if (rank <= 20) return "bg-orange-500/15 text-orange-700 dark:text-orange-400";
   193→  return "bg-rose-500/15 text-rose-700 dark:text-rose-400";
   194→}

   259→    queryKey: ["seo", "comparison"],
   260→    queryFn: () => api<LocationComparisonRow[]>("/api/seo/location-comparison"),
   261→  });

   264→    queryKey: ["seo-audits", activeLocationId],
   265→    queryFn: () => {
   266→      const url = activeLocationId && activeLocationId !== "all"
   267→        ? `/api/seo-audits?locationId=${encodeURIComponent(activeLocationId)}`
   268→        : "/api/seo-audits";

   270→    },

   272→
   273→  const competitorsQuery = useQuery<CompetitorRow[]>({
   274→    queryKey: ["competitors", activeLocationId],

   286→  const isKeywordsLoading = keywordsQuery.isLoading;
   287→

   289→  useEffect(() => {
   290→    if (keywords.length > 0) {
   291→      if (!gridKeywordId || !keywords.find((k) => k.id === gridKeywordId)) {
   292→        const best = [...keywords].sort(
   293→          (a, b) => (a.currentRank ?? 99) - (b.currentRank ?? 99),

   295→        setGridKeywordId(best.id);
   296→      }

   301→

   304→    setSelectedKeywordId(null);

   306→

   311→      const loc = locQuery.data.find((l) => l.id === activeLocationId);
   312→      return loc ? loc.healthScore : null;
   313→    }
   314→    const avg = locQuery.data.reduce((s, l) => s + l.healthScore, 0) / locQuery.data.length;
   315→    return Math.round(avg);
   316→  }, [locQuery.data, activeLocationId]);
   317→
   318→  const visibilityScore = useMemo(() => {
   319→    if (!locQuery.data || locQuery.data.length === 0) return null;
   320→    if (activeLocationId && activeLocationId !== "all") {
   321→      const loc = locQuery.data.find((l) => l.id === activeLocationId);
   322→      return loc ? loc.visibilityScore : null;
   323→    }
   324→    const avg = locQuery.data.reduce((s, l) => s + l.visibilityScore, 0) / locQuery.data.length;
   325→    return Math.round(avg);
   326→  }, [locQuery.data, activeLocationId]);
   327→

   329→    if (!activeLocationId || activeLocationId === "all") return "All locations";

   346→      queryClient.invalidateQueries({ queryKey: ["competitors"] });

   348→      const msg = e instanceof Error ? e.message : "Failed to refresh rankings";
   349→      toast.error(msg);
   350→    } finally {
   351→      setRefreshing(false);
   352→    }
   353→  }, [activeLocationId, queryClient]);

   362→        method: "POST",
   363→        body: JSON.stringify({ action: "seo", locationId: activeLocationId }),

   369→      const msg = e instanceof Error ? e.message : "Failed to fetch recommendations";
   370→      setRecsError(msg);
   371→      toast.error(msg);
   372→    } finally {
   373→      setRecsLoading(false);
   374→    }
   375→  }, [aiDisabled, activeLocationId]);
   376→
   377→  const fetchSummary = useCallback(async () => {

   380→    setSummaryError(null);

   383→      const res = await api<{ summary: string }>("/api/ai", {
   384→        method: "POST",
   385→        body: JSON.stringify({ action: "summary", locationId: activeLocationId }),

   388→      toast.success("MiSA AI monthly summary ready");

   391→      const msg = e instanceof Error ? e.message : "Failed to fetch summary";

   992→        {label}

  1003→function RankChangeBadge({ change }: { change: number }) {
  1004→  if (change === 0) {
  1005→    return (
  1006→      <span className="inline-flex items-center gap-0.5 text-xs text-muted-foreground tabular-nums">
  1007→        <Minus className="size-3" /> 0
  1008→      </span>
  1009→    );
  1010→  }
  1011→  const isGood = change > 0; // positive rankChange = improvement (rank went down)
  1012→  return (
  1013→    <span
  1014→      className={cn(
  1015→        "inline-flex items-center gap-0.5 text-xs font-medium tabular-nums",
  1016→        isGood ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400",
  1017→      )}
  1018→    >
  1019→      {isGood ? <ArrowUp className="size-3" /> : <ArrowDown className="size-3" />}
  1020→      {Math.abs(change)}
  1021→    </span>
  1022→  );
  1023→}

  1025→function MiniSparkline({ history }: { history: { rank: number; date: string }[] }) {
  1026→  const data = useMemo(() => {
  1027→    return (history ?? [])
  1028→      .filter((h) => h.rank > 0)
  1029→      .slice()
  1030→      .reverse() // API returns newest first; chart wants oldest first
  1031→      .map((h) => ({ rank: h.rank }));
  1032→  }, [history]);
  1033→
  1034→  if (data.length < 2) {
  1035→    return <span className="text-xs text-muted-foreground">—</span>;
  1036→  }
  1037→  const first = data[0].rank;
  1038→  const last = data[data.length - 1].rank;
  1039→  const delta = first - last; // positive = improvement
  1040→  const color = delta > 0 ? "var(--chart-1)" : delta < 0 ? "var(--chart-4)" : "var(--muted-foreground)";
  1041→  return (
  1042→    <div className="w-16 h-6 inline-block">
  1043→      <ResponsiveContainer width="100%" height="100%">
  1044→        <LineChart data={data} margin={{ top: 1, right: 1, bottom: 1, left: 1 }}>
  1045→          <YAxis hide domain={["dataMin - 1", "dataMax + 1"]} reversed />
  1046→          <Line type="monotone" dataKey="rank" stroke={color} strokeWidth={1.5} dot={false} isAnimationActive={false} />
  1047→        </LineChart>
  1048→      </ResponsiveContainer>
  1049→    </div>
  1050→  );
  1051→}
