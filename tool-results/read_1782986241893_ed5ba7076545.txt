    84→/* ---------- Helpers ---------- */
    85→
    86→function postTypeLabel(t: string): string {
    87→  return TYPE_META[t as PostType]?.label ?? t;
    88→}
    89→
    90→function postTypeMeta(t: string): TypeMeta {
    91→  return TYPE_META[t as PostType] ?? TYPE_META.update;
    92→}
    93→
    94→function relativeTime(iso: string | null): string | null {
    95→  if (!iso) return null;
    96→  try {
    97→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
    98→  } catch {
    99→    return null;
   100→  }
   101→}
   102→
   103→function scheduleLabel(iso: string | null): string {
   104→  if (!iso) return "";
   105→  try {
   106→    const d = new Date(iso);
   107→    if (isToday(d)) return `Today, ${format(d, "h:mm a")}`;
   108→    if (isTomorrow(d)) return `Tomorrow, ${format(d, "h:mm a")}`;
   109→    return format(d, "d MMM yyyy, h:mm a");
   110→  } catch {
   111→    return "";
   112→  }
   113→}

   178→    setEditorOpen(true);

   189→        method: "PATCH",
   190→        body: JSON.stringify({ status: "published" }),

   192→      qc.invalidateQueries({ queryKey: ["posts"] });

   205→      setDeletingPost(null);

   207→      toast.error(e?.message || "Failed to delete", { id: `del-${p.id}` });
   208→    }

   211→  function handleSaved() {
   212→    qc.invalidateQueries({ queryKey: ["posts"] });

   218→      <PageHeader
   219→        title="Google Posts"

   223→          <>
   224→            <Select value={activeLocationId} onValueChange={(v) => setActiveLocationId(v as any)}>

   227→                <SelectValue placeholder="All locations" />
   228→              </SelectTrigger>
   229→              <SelectContent>
   230→                <SelectItem value="all">All locations</SelectItem>

   233→                    {l.name} — {l.city}
   234→                  </SelectItem>
   235→                ))}
   236→              </SelectContent>
   237→            </Select>
   238→            {canManage && (
   239→              <Button size="sm" onClick={openCreate}>
   240→                <Plus className="size-3.5 mr-1.5" /> New post
   241→              </Button>
   242→            )}
   243→          </>
   244→        }
   245→      />
   246→
   247→      {/* Stat row */}
   248→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3 sm:gap-4">
   249→        {isLoading ? (
   250→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)
   251→        ) : (
   252→          <>
   253→            <StatCard label="Published"    value={stats.published}   icon={CheckCircle2} accent="emerald" hint="Live on Google" />
   254→            <StatCard label="Scheduled"    value={stats.scheduled}   icon={CalendarClock} accent="amber"   hint="Queued for later" />
   255→            <StatCard label="Drafts"       value={stats.drafts}      icon={FileText}     accent="slate"   hint="Not yet published" />
   256→            <StatCard label="AI-Generated" value={stats.aiGenerated} icon={Sparkles}     accent="amber"   hint="By MiSA AI" />

   258→        )}
   259→      </div>
   260→
   261→      {/* Filter bar */}
   262→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   263→        <Tabs value={statusFilter} onValueChange={(v) => setStatusFilter(v as StatusFilter)}>
   264→          <TabsList>
   265→            <TabsTrigger value="all">

   267→              {statsSource.length > 0 && (
   268→                <span className="ml-1 text-[10px] text-muted-foreground tabular-nums">
   269→                  {statsSource.length}
   270→                </span>
   271→              )}
   272→            </TabsTrigger>
   273→            <TabsTrigger value="published">Published</TabsTrigger>
   274→            <TabsTrigger value="scheduled">Scheduled</TabsTrigger>
   275→            <TabsTrigger value="draft">Drafts</TabsTrigger>
   276→          </TabsList>
   277→        </Tabs>
   278→
   279→        <Select value={typeFilter} onValueChange={(v) => setTypeFilter(v as TypeFilter)}>
   280→          <SelectTrigger size="sm" className="w-full sm:w-[180px]">
   281→            <SelectValue placeholder="All types" />
   282→          </SelectTrigger>
   283→          <SelectContent>
   284→            <SelectItem value="all">All types</SelectItem>
   285→            <SelectItem value="whats_new">What's New</SelectItem>
   286→            <SelectItem value="offer">Offer</SelectItem>
   287→            <SelectItem value="event">Event</SelectItem>
   288→            <SelectItem value="update">Update</SelectItem>
   289→          </SelectContent>
   290→        </Select>
   291→      </div>
   292→
   293→      {/* Posts grid */}
   294→      {isLoading ? (
   295→        <PostsGridSkeleton />

   305→                canManage={canManage}

   307→                onEdit={openEdit}

   310→            ))}

   318→        onOpenChange={(o) => (o ? setEditorOpen(true) : closeEditor())}

   320→        locations={locations ?? []}
   321→        defaultLocationId={activeLocationId !== "all" ? activeLocationId : undefined}
   322→        onSaved={handleSaved}
   323→      />
   324→
   325→      {/* Delete confirm */}
   326→      <AlertDialog open={!!deletingPost} onOpenChange={(o) => !o && setDeletingPost(null)}>
   327→        <AlertDialogContent>
   328→          <AlertDialogHeader>
   329→            <AlertDialogTitle>Delete this post?</AlertDialogTitle>
   330→            <AlertDialogDescription>

   335→                </>
   336→              )}
   337→            </AlertDialogDescription>
   338→          </AlertDialogHeader>
   339→          <AlertDialogFooter>
   340→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   341→            <AlertDialogAction
   342→              className="bg-destructive text-white hover:bg-destructive/90"
   343→              onClick={() => deletingPost && deletePost(deletingPost)}

   345→              <Trash2 className="size-3.5 mr-1.5" /> Delete post
   346→            </AlertDialogAction>
   347→          </AlertDialogFooter>
   348→        </AlertDialogContent>
   349→      </AlertDialog>
   350→    </div>
   351→  );
   352→}

   354→/* ---------- Post card ---------- */
   355→
   356→function PostCard({
   357→  post, canManage, onPublish, onEdit, onDelete,
   358→}: {
   359→  post: PostWithLocation;
   360→  canManage: boolean;
   361→  onPublish: (p: PostWithLocation) => void;
   362→  onEdit: (p: PostWithLocation) => void;
   363→  onDelete: (p: PostWithLocation) => void;
   364→}) {
   365→  const meta = postTypeMeta(post.type);
   366→  const Icon = meta.icon;
   367→  const [scheduleOpen, setScheduleOpen] = React.useState(false);
   368→  const [scheduleDate, setScheduleDate] = React.useState<Date | null>(
   369→    post.scheduledAt ? new Date(post.scheduledAt) : null,
   370→  );
   371→  const [scheduling, setScheduling] = React.useState(false);
   372→  const qc = useQueryClient();
   373→
   374→  const relPublished = relativeTime(post.publishedAt);
   375→  const relScheduled = relativeTime(post.scheduledAt);
   376→  const relCreated = relativeTime(post.createdAt);
   377→
   378→  async function applySchedule() {
   379→    if (!scheduleDate) return;
   380→    setScheduling(true);
   381→    try {
   382→      await api(`/api/posts/${post.id}`, {
   383→        method: "PATCH",
   384→        body: JSON.stringify({ status: "scheduled", scheduledAt: scheduleDate.toISOString() }),
   385→      });
   386→      qc.invalidateQueries({ queryKey: ["posts"] });
   387→      toast.success(`Scheduled for ${scheduleLabel(scheduleDate.toISOString())}`);
   388→      setScheduleOpen(false);
   389→    } catch (e: any) {
   390→      toast.error(e?.message || "Failed to schedule");
   391→    } finally {
   392→      setScheduling(false);
   393→    }
   394→  }
   395→
   396→  return (
   397→    <Card className="group relative hover:shadow-md transition-shadow overflow-hidden">
   398→      <CardContent className="p-4 space-y-3">
   399→        {/* Header: type + actions */}
   400→        <div className="flex items-start justify-between gap-2">
   401→          <div className="flex items-center gap-2 min-w-0">
   402→            <span className={cn("size-8 rounded-lg flex items-center justify-center shrink-0", meta.tint)}>
   403→              <Icon className="size-4" />
   404→            </span>
   405→            <div className="min-w-0">
   406→              <div className="text-[11px] font-medium text-muted-foreground uppercase tracking-wide">
   407→                {meta.label}
   408→              </div>
   409→              <div className="text-[10px] text-muted-foreground truncate flex items-center gap-1">
   410→                <MapPin className="size-2.5" />
   411→                {post.locationName}
   412→              </div>
   413→            </div>
   414→          </div>
   415→
   416→          {canManage && (
   417→            <DropdownMenu>
   418→              <DropdownMenuTrigger asChild>
   419→                <Button variant="ghost" size="icon" className="size-7 -mt-1 -mr-1 opacity-60 group-hover:opacity-100 transition-opacity">
   420→                  <MoreVertical className="size-4" />
   421→                </Button>
   422→              </DropdownMenuTrigger>
   423→              <DropdownMenuContent align="end" className="w-44">
   424→                {(post.status === "draft" || post.status === "scheduled" || post.status === "failed") && (
   425→                  <DropdownMenuItem onClick={() => onPublish(post)}>
   426→                    <Send className="size-3.5 mr-2 text-emerald-500" /> Publish now
   427→                  </DropdownMenuItem>
   428→                )}
   429→                <DropdownMenuItem onClick={() => setScheduleOpen(true)}>
   430→                  <CalendarClock className="size-3.5 mr-2 text-amber-500" /> Schedule…
   431→                </DropdownMenuItem>
   432→                <DropdownMenuItem onClick={() => onEdit(post)}>
   433→                  <Pencil className="size-3.5 mr-2" /> Edit
   434→                </DropdownMenuItem>
   435→                <DropdownMenuSeparator />
   436→                <DropdownMenuItem
   437→                  className="text-rose-600 dark:text-rose-400 focus:text-rose-700"
   438→                  onClick={() => onDelete(post)}
   439→                >
   440→                  <Trash2 className="size-3.5 mr-2" /> Delete
   441→                </DropdownMenuItem>
   442→              </DropdownMenuContent>
   443→            </DropdownMenu>
   444→          )}
   445→        </div>
   446→
   447→        {/* Title */}
   448→        <h3 className="font-semibold text-sm leading-snug line-clamp-1">{post.title || "Untitled post"}</h3>
   449→
   450→        {/* Content preview */}
   451→        <p className="text-xs text-muted-foreground line-clamp-3 leading-relaxed">
   452→          {post.content}
   453→        </p>
   454→
   455→        {/* Status + AI badge */}
   456→        <div className="flex items-center gap-1.5 flex-wrap">
   457→          <PostStatusBadge status={post.status} />
   458→          {post.source === "ai" && (
   459→            <Badge
   460→              variant="outline"
   461→              className="text-[10px] py-0 px-1.5 gap-0.5 bg-amber-500/10 text-amber-600 dark:text-amber-400 border-amber-500/20 font-medium"
   462→            >
   463→              <Sparkles className="size-2.5" />
   464→              MiSA AI
   465→            </Badge>
   466→          )}
   467→          {post.ctaType && (
   468→            <Badge variant="outline" className="text-[10px] py-0 px-1.5 font-normal">
   469→              <ArrowRight className="size-2.5 mr-0.5" />
   470→              {CTA_LABEL[post.ctaType] ?? post.ctaType}
   471→            </Badge>
   472→          )}
   473→        </div>
   474→
   475→        {/* Footer: time + CTA */}
   476→        <div className="flex items-center justify-between gap-2 pt-2 border-t text-[10px] text-muted-foreground">
   477→          <span className="inline-flex items-center gap-1 min-w-0 truncate">
   478→            {post.status === "published" && relPublished ? (
   479→              <>
   480→                <CheckCircle2 className="size-2.5 text-emerald-500 shrink-0" />
   481→                <span className="truncate">Published {relPublished}</span>
   482→              </>
   483→            ) : post.status === "scheduled" && relScheduled ? (
   484→              <>
   485→                <Clock className="size-2.5 text-amber-500 shrink-0" />
   486→                <span className="truncate">Goes live {relScheduled}</span>
   487→              </>
   488→            ) : (
   489→              <>
   490→                <Clock className="size-2.5 shrink-0" />
   491→                <span className="truncate">Created {relCreated}</span>
   492→              </>
   493→            )}
   494→          </span>
   495→          {post.ctaUrl && (
   496→            <a
   497→              href={post.ctaUrl}
   498→              target="_blank"
   499→              rel="noopener noreferrer"
   500→              className="inline-flex items-center gap-0.5 text-primary hover:underline shrink-0"
   501→              onClick={(e) => e.stopPropagation()}
   502→            >
   503→              <ExternalLink className="size-2.5" /> Link
   504→            </a>
   505→          )}
   506→        </div>
   507→      </CardContent>
   508→
   509→      {/* Schedule popover */}
   510→      <Dialog open={scheduleOpen} onOpenChange={(o) => setScheduleOpen(o)}>
   511→        <DialogContent className="sm:max-w-md">
   512→          <DialogHeader>
   513→            <DialogTitle>Schedule post</DialogTitle>
   514→            <DialogDescription>
   515→              Pick when this post should go live on Google Business Profile.
   516→            </DialogDescription>
   517→          </DialogHeader>
   518→          <div className="flex flex-col items-center gap-3">
   519→            <Popover>
   520→              <PopoverTrigger asChild>
   521→                <Button variant="outline" className="w-full justify-start text-left font-normal">
   522→                  <CalendarClock className="size-4 mr-2 text-amber-500" />
   523→                  {scheduleDate ? scheduleLabel(scheduleDate.toISOString()) : "Pick a date & time"}
   524→                </Button>
   525→              </PopoverTrigger>
   526→              <PopoverContent className="w-auto p-0">
