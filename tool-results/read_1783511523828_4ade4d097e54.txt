    93→    93→export async function getValidAccessToken(): Promise<string | null> {
    94→    94→  const account = await db.googleAccount.findFirst();
    95→    95→  if (!account) return null;
    96→    96→
    97→    97→  // Check if token is still valid (5 min buffer)
    98→    98→  if (account.tokenExpiry && new Date(account.tokenExpiry) > new Date(Date.now() + 5 * 60 * 1000)) {
    99→    99→    return account.accessToken;
   100→   100→  }
   101→   101→
   102→   102→  // Try to refresh
   103→   103→  if (account.refreshToken) {
   104→   104→    try {
   105→   105→      const { accessToken, expiryDate } = await refreshAccessToken(account.refreshToken);
   106→   106→      await db.googleAccount.update({
   107→   107→        where: { id: account.id },
   108→   108→        data: { accessToken, tokenExpiry: new Date(expiryDate), status: "active" },
   109→   109→      });
   110→   110→      return accessToken;
   111→   111→    } catch (e) {
   112→   112→      console.error("Token refresh failed:", e);
   113→   113→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   114→   114→      return null;
   115→   115→    }
   116→   116→  }
   117→   117→
   118→   118→  return null;
   119→   119→}

   208→   208→// Fetch ALL 5 metric types for a date range and return as daily aggregates
   209→   209→export async function getFullPerformanceMetrics(
   210→   210→  accessToken: string,
   211→   211→  locationName: string,
   212→   212→  daysBack: number = 30
   213→   213→): Promise<{ date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }[]> {
   214→   214→  const today = new Date();
   215→   215→  const start = new Date(today);
   216→   216→  start.setDate(today.getDate() - daysBack);
   217→   217→
   218→   218→  const startDate = { year: start.getFullYear(), month: start.getMonth() + 1, day: start.getDate() };
   219→   219→  const endDate = { year: today.getFullYear(), month: today.getMonth() + 1, day: today.getDate() };
   220→   220→
   221→   221→  // Google API returns one metric type per call, so fetch all 5 in parallel
   222→   222→  const metricTypes = {
   223→   223→    searchViews: "BUSINESS_IMPRESSIONS_DESKTOP_SEARCH",
   224→   224→    mapsViews: "BUSINESS_IMPRESSIONS_DESKTOP_MAPS",
   225→   225→    websiteClicks: "WEBSITE_CLICKS",
   226→   226→    phoneCalls: "CALL_CLICKS",
   227→   227→    directionRequests: "BUSINESS_DIRECTION_REQUESTS",
   228→   228→  };
   229→   229→
   230→   230→  const [searchRes, mapsRes, websiteRes, callRes, directionRes] = await Promise.allSettled([
   231→   231→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.searchViews),
   232→   232→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.mapsViews),
   233→   233→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.websiteClicks),
   234→   234→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.phoneCalls),
   235→   235→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.directionRequests),
   236→   236→  ]);
   237→   237→
   238→   238→  // Build a map of date → metrics
   239→   239→  const dailyMap = new Map<string, { date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }>();
   240→   240→
   241→   241→  function populate(result: PromiseSettledResult<any[]>, key: keyof typeof metricTypes) {
   242→   242→    if (result.status !== "fulfilled") return;
   243→   243→    for (const entry of result.value) {
   244→   244→      const dateStr = entry.date?.year
   245→   245→        ? `${entry.date.year}-${String(entry.date.month).padStart(2, "0")}-${String(entry.date.day).padStart(2, "0")}`
   246→   246→        : null;
   247→   247→      if (!dateStr) continue;
   248→   248→      if (!dailyMap.has(dateStr)) {
   249→   249→        dailyMap.set(dateStr, {
   250→   250→          date: new Date(dateStr),
   251→   251→          searchViews: 0, mapsViews: 0, websiteClicks: 0, phoneCalls: 0, directionRequests: 0,
   252→   252→        });
   253→   253→      }
   254→   254→      const day = dailyMap.get(dateStr)!;
   255→   255→      const value = entry.value ?? 0;
   256→   256→      (day as any)[key] += value;
   257→   257→    }
   258→   258→  }
   259→   259→
   260→   260→  populate(searchRes, "searchViews");
   261→   261→  populate(mapsRes, "mapsViews");
   262→   262→  populate(websiteRes, "websiteClicks");
   263→   263→  populate(callRes, "phoneCalls");
   264→   264→  populate(directionRes, "directionRequests");
   265→   265→
   266→   266→  return Array.from(dailyMap.values()).sort((a, b) => a.date.getTime() - b.date.getTime());
   267→   267→}

   319→   319→export async function updateGoogleBusinessProfile(
   320→   320→  accessToken: string,
   321→   321→  locationName: string,
   322→   322→  updates: {
   323→   323→    title?: string;
   324→   324→    phone?: string;
   325→   325→    website?: string;
   326→   326→    description?: string;
   327→   327→    appointmentUrl?: string;
   328→   328→    hours?: { dayOfWeek: string; openTime?: { hours: number; minutes: number }; closeTime?: { hours: number; minutes: number }; isClosed?: boolean }[];
   329→   329→    categories?: { primaryCategory: string; additionalCategories?: string[] };
   330→   330→  }
   331→   331→): Promise<any> {
   332→   332→  const body: any = {};
   333→   333→  const fieldMask: string[] = [];
   334→   334→
   335→   335→  if (updates.title !== undefined) { body.title = updates.title; fieldMask.push("title"); }
   336→   336→  if (updates.phone !== undefined) { body.phoneNumbers = { primaryPhone: updates.phone }; fieldMask.push("phoneNumbers"); }
   337→   337→  if (updates.website !== undefined) { body.websiteUri = updates.website; fieldMask.push("websiteUri"); }
   338→   338→  if (updates.description !== undefined) {
   339→   339→    body.profile = { ...(body.profile || {}), description: updates.description };
   340→   340→    fieldMask.push("profile.description");
   341→   341→  }

   350→   350→        openTime: h.openTime,

   352→   352→        closeTime: h.closeTime,
   353→   353→      })),
   354→   354→    };

   356→   356→  }
   357→   357→  if (updates.categories) {
   358→   358→    body.categories = {
   359→   359→      primaryCategory: { displayName: updates.categories.primaryCategory },
   360→   360→      additionalCategories: (updates.categories.additionalCategories || []).map(c => ({ displayName: c })),
   361→   361→    };
   362→   362→    fieldMask.push("categories");
   363→   363→  }
   364→   364→
   365→   365→  const res = await fetch(`${GBP_API_BASE}/${locationName}?updateMask=${fieldMask.join(",")}&validateOnly=false`, {
   366→   366→    method: "PATCH",
   367→   367→    headers: {
   368→   368→      Authorization: `Bearer ${accessToken}`,
   369→   369→      "Content-Type": "application/json",
   370→   370→    },
   371→   371→    body: JSON.stringify(body),
   372→   372→  });
   373→   373→
   374→   374→  if (!res.ok) {
   375→   375→    const errText = await res.text();
   376→   376→    throw new Error(`Failed to update Google Business Profile: ${res.status} ${errText}`);
   377→   377→  }
   378→   378→  return res.json();
   379→   379→}
