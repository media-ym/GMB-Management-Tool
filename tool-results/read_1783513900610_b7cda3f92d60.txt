   160→function relativeTime(iso: string | null): string {
   161→  if (!iso) return "Never";
   162→  try {
   163→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
   164→  } catch {
   165→    return "—";
   166→  }
   167→}
   168→
   169→function fullTime(iso: string | null): string {
   170→  if (!iso) return "—";
   171→  try {
   172→    return format(new Date(iso), "d MMM yyyy, h:mm a");
   173→  } catch {
   174→    return "—";
   175→  }
   176→}
   177→
   178→function scopeLabel(scope: string): string {
   179→  const found = REQUESTED_SCOPES.find((s) => s.scope === scope);
   180→  return found?.label ?? scope.replace(/^https:\/\/www\.googleapis\.com\/auth\/business\./, "").replace(/^https:\/\/www\.googleapis\.com\/auth\/userinfo\./, "userinfo.").replace(/^https:\/\/www\.googleapis\.com\/auth\//, "");
   181→}
   182→
   183→function tokenProgress(expiryIso: string | null): { pct: number; totalMs: number; remainingMs: number } {
   184→  if (!expiryIso) return { pct: 0, totalMs: 0, remainingMs: 0 };
   185→  const now = Date.now();
   186→  const expiry = new Date(expiryIso).getTime();
   187→  // assume access tokens are issued with a 1h lifetime
   188→  const totalMs = 60 * 60 * 1000;
   189→  const remainingMs = expiry - now;
   190→  const pct = Math.max(0, Math.min(100, (remainingMs / totalMs) * 100));
   191→  return { pct, totalMs, remainingMs };
   192→}

   258→function CopyButton({ value, label = "Copy" }: { value: string; label?: string }) {
   259→  const [copied, setCopied] = useState(false);
   260→  async function onCopy() {
   261→    try {
   262→      await navigator.clipboard.writeText(value);
   263→      setCopied(true);
   264→      toast.success("Copied to clipboard");
   265→      setTimeout(() => setCopied(false), 1500);
   266→    } catch {
   267→      toast.error("Couldn't copy to clipboard");
   268→    }
   269→  }
   270→  return (
   271→    <Button
   272→      type="button"
   273→      variant="ghost"
   274→      size="sm"
   275→      onClick={onCopy}
   276→      className="h-7 px-2 text-xs text-muted-foreground hover:text-foreground"
   277→    >
   278→      {copied ? <Check className="size-3.5 text-emerald-500" /> : <Copy className="size-3.5" />}
   279→      {copied ? "Copied" : label}
   280→    </Button>
   281→  );
   282→}
