   167→   167→// Tracks per-client authorization for managing their Google Business Profile.
   168→   168→// Google's Third-Party Policy requires explicit authorization before managing
   169→   169→// GBP on behalf of an end-client. Each GBP write operation (reply, post, profile update)
   170→   170→// must verify authorization.status === "active".
   171→   171→model ClientAuthorization {
   172→   172→  id                String    @id @default(cuid())
   173→   173→  clientId          String
   174→   174→  // What the client has authorized us to do on their behalf
   175→   175→  authorizedScopes  String?   // JSON array: ["review.reply", "post.create", "profile.update", "analytics.sync"]
   176→   176→  // Authorization lifecycle
   177→   177→  status            String    @default("active") // active | revoked | expired
   178→   178→  grantedAt         DateTime  @default(now())
   179→   179→  revokedAt         DateTime?
   180→   180→  expiresAt         DateTime? // nullable = no expiry
   181→   181→  // Audit trail
   182→   182→  grantedByUserId   String?   // Which MyFNG user recorded the authorization
   183→   183→  authorizationDoc  String?   // URL or path to signed authorization document
   184→   184→  notes             String?
   185→   185→  createdAt         DateTime  @default(now())
   186→   186→  updatedAt         DateTime  @updatedAt
   187→   187→
   188→   188→  client            Client    @relation(fields: [clientId], references: [id], onDelete: Cascade)
   189→   189→
   190→   190→  @@index([clientId])
   191→   191→  @@index([status])
   192→   192→}
