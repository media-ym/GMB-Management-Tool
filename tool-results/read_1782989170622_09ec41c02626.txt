    34→// ---- Types --------------------------------------------------------------
    35→
    36→interface AnalyticsTotals {
    37→  searchViews: number;
    38→  mapsViews: number;
    39→  websiteClicks: number;
    40→  phoneCalls: number;
    41→  directionRequests: number;
    42→}
    43→
    44→interface PerLocationRow {
    45→  locationId: string;
    46→  name: string;
    47→  city: string;

    51→interface AnalyticsResponse {
    52→  series: AnalyticsPoint[];

    56→
    57→type MetricKey = keyof AnalyticsTotals;
    58→type SortKey = MetricKey | "name" | "total";

    60→// ---- Helpers ------------------------------------------------------------
    61→
    62→const fmt = (n: number) =>
    63→  n >= 1_000_000 ? `${(n / 1_000_000).toFixed(1)}M` :
    64→  n >= 1_000 ? `${(n / 1_000).toFixed(1)}k` : `${n}`;

    67→  searchViews: "var(--chart-1)",      // emerald
    68→  mapsViews: "var(--chart-2)",        // amber
    69→  websiteClicks: "var(--chart-3)",    // teal

    71→  directionRequests: "var(--chart-5)",// cyan
    72→};

    75→  searchViews: "#10b981",
    76→  mapsViews: "#f59e0b",
    77→  websiteClicks: "#14b8a6",
    78→  phoneCalls: "#f43f5e",
    79→  directionRequests: "#06b6d4",
    80→};
    81→
    82→/** % change of second half of series vs first half. */
    83→function computeDelta(series: AnalyticsPoint[], key: MetricKey): number | undefined {
    84→  if (series.length < 4) return undefined;
    85→  const mid = Math.floor(series.length / 2);
    86→  const first = series.slice(0, mid).reduce((s, p) => s + p[key], 0);
    87→  const second = series.slice(mid).reduce((s, p) => s + p[key], 0);
    88→  if (first === 0) return undefined;
    89→  return Math.round(((second - first) / first) * 100);
    90→}

   170→      .slice(0, 10);

   172→  }, [perLocation]);

   175→  const tableRows = useMemo(() => {

   177→    rows.sort((a, b) => {
   178→      let cmp = 0;

   180→      else if (sortKey === "total") cmp = a.total - b.total;
   181→      else cmp = (a[sortKey] ?? 0) - (b[sortKey] ?? 0);

   184→    return rows;

   186→

   191→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));

   193→      setSortKey(key);
   194→      setSortDir(key === "name" ? "asc" : "desc");
   195→    }

   197→
   198→  return (

   200→      <PageHeader
   201→        title="Analytics"
   202→        description="Google Business Profile performance across all locations"
   203→        icon={BarChart3}
   204→        actions={

   206→            <Select value={activeLocationId} onValueChange={(v) => setActiveLocationId(v as string | "all")}>
   207→              <SelectTrigger size="sm" className="w-[180px] sm:w-[220px]">
   208→                <Filter className="size-3.5 mr-1.5 text-muted-foreground" />

   210→              </SelectTrigger>
   211→              <SelectContent>
   212→                <SelectItem value="all">All locations</SelectItem>
   213→                {(locations ?? []).map((l) => (

   215→                    {l.name} · {l.city}
   216→                  </SelectItem>

   218→              </SelectContent>
   219→            </Select>
   220→
   221→            <Select value={String(days)} onValueChange={(v) => setDays(Number(v))}>
   222→              <SelectTrigger size="sm" className="w-[130px]">
   223→                <Calendar className="size-3.5 mr-1.5 text-muted-foreground" />
   224→                <SelectValue />
   225→              </SelectTrigger>
   226→              <SelectContent>
   227→                <SelectItem value="7">7 days</SelectItem>
   228→                <SelectItem value="30">30 days</SelectItem>
   229→                <SelectItem value="90">90 days</SelectItem>
   230→              </SelectContent>
   231→            </Select>

   239→          Array.from({ length: 5 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)

   247→              hint={`Last ${days} days`}
   248→              delta={computeDelta(series, "searchViews")}
   249→              deltaLabel="vs prior half"
   250→            />
   251→            <StatCard
   252→              label="Maps Views"
   253→              value={fmt(totals.mapsViews)}
   254→              icon={Map}
   255→              accent="amber"
   256→              hint={`Last ${days} days`}
   257→              delta={computeDelta(series, "mapsViews")}
   258→              deltaLabel="vs prior half"
   259→            />
   260→            <StatCard
   261→              label="Website Clicks"
   262→              value={fmt(totals.websiteClicks)}
   263→              icon={MousePointerClick}
   264→              accent="teal"
   265→              hint={`Last ${days} days`}
   266→              delta={computeDelta(series, "websiteClicks")}
   267→              deltaLabel="vs prior half"
   268→            />
   269→            <StatCard
   270→              label="Phone Calls"
   271→              value={fmt(totals.phoneCalls)}
   272→              icon={Phone}
   273→              accent="rose"
   274→              hint={`Last ${days} days`}

   282→              accent="emerald"

   316→                    <AreaChart data={trendData} margin={{ top: 5, right: 5, left: -20, bottom: 0 }}>

   320→                          <stop offset="95%" stopColor={METRIC_COLORS.searchViews} stopOpacity={0} />

   322→                        <linearGradient id="aMaps" x1="0" y1="0" x2="0" y2="1">

   324→                          <stop offset="95%" stopColor={METRIC_COLORS.mapsViews} stopOpacity={0} />
   325→                        </linearGradient>
   326→                      </defs>
   327→                      <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" vertical={false} />

   329→                      <YAxis tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" tickFormatter={fmt} />
   330→                      <Tooltip contentStyle={tooltipStyle} labelStyle={tooltipLabelStyle} formatter={(v: number) => fmt(v)} />

   333→                      <Area type="monotone" dataKey="maps" name="Maps Views" stroke={METRIC_COLORS.mapsViews} strokeWidth={2} fill="url(#aMaps)" />

   335→                  </ResponsiveContainer>
   336→                )}
   337→              </div>
   338→            </CardSection>
   339→
   340→            <CardSection
   341→              title="Engagement Breakdown"
   342→              description="Clicks · Calls · Directions"
   343→            >
   344→              <div className="h-[240px]">
   345→                {isLoading || engagementData.length === 0 ? (
   346→                  <Skeleton className="h-full w-full" />
   347→                ) : (
   348→                  <ResponsiveContainer width="100%" height="100%">
   349→                    <PieChart>
   350→                      <Pie
   351→                        data={engagementData}
   352→                        dataKey="value"
   353→                        nameKey="name"
   354→                        cx="50%"
   355→                        cy="50%"
   356→                        innerRadius={45}
   357→                        outerRadius={80}
   358→                        paddingAngle={2}
   359→                        stroke="var(--card)"
   360→                        strokeWidth={2}
   361→                      >
   362→                        {engagementData.map((d) => (
   363→                          <Cell key={d.key} fill={METRIC_COLORS[d.key]} />
   364→                        ))}
   365→                        <LabelList
   366→                          dataKey="value"
   367→                          position="outside"
   368→                          formatter={(v: number) => fmt(v)}
   369→                          style={{ fontSize: 10, fill: "var(--muted-foreground)" }}
   370→                        />
   371→                      </Pie>
   372→                      <Tooltip contentStyle={tooltipStyle} formatter={(v: number) => fmt(v)} />
   373→                      <Legend wrapperStyle={{ fontSize: 11 }} />
   374→                    </PieChart>
   375→                  </ResponsiveContainer>
   376→                )}
   377→              </div>
   378→            </CardSection>
   379→          </div>
   380→

   383→            <CardSection
   384→              title="Top Locations by Search Views"

   386→              className="lg:col-span-2"
   387→              action={<Badge variant="outline" className="text-xs">Top {Math.min(10, locationBars.length)}</Badge>}
   388→            >
   389→              <div className="h-[300px]">
   390→                {isLoading || locationBars.length === 0 ? (
   391→                  <Skeleton className="h-full w-full" />

   393→                  <ResponsiveContainer width="100%" height="100%">
   394→                    <BarChart
   395→                      data={locationBars}
   396→                      layout="vertical"
   397→                      margin={{ top: 0, right: 24, left: 0, bottom: 0 }}
   398→                      barCategoryGap={8}
   399→                    >
   400→                      <defs>
   401→                        <linearGradient id="gLocation" x1="0" y1="0" x2="1" y2="0">
   402→                          <stop offset="0%" stopColor={METRIC_HEX.searchViews} stopOpacity={0.5} />
   403→                          <stop offset="100%" stopColor={METRIC_HEX.searchViews} stopOpacity={1} />
   404→                        </linearGradient>
   405→                      </defs>
   406→                      <CartesianGrid strokeDasharray="3 3" stroke="var(--border)" horizontal={false} />
   407→                      <XAxis type="number" tick={{ fontSize: 11 }} stroke="var(--muted-foreground)" tickFormatter={fmt} />
   408→                      <YAxis
   409→                        type="category"
   410→                        dataKey="city"
   411→                        tick={{ fontSize: 11 }}
   412→                        stroke="var(--muted-foreground)"
   413→                        width={88}
   414→                        interval={0}
   415→                      />
   416→                      <Tooltip
   417→                        contentStyle={tooltipStyle}
   418→                        labelStyle={tooltipLabelStyle}
   419→                        formatter={(v: number) => [fmt(v), "Search Views"]}
   420→                      />
   421→                      <Bar dataKey="searchViews" name="Search Views" fill="url(#gLocation)" radius={[0, 4, 4, 0]} maxBarSize={20}>
   422→                        <LabelList
   423→                          dataKey="searchViews"
   424→                          position="right"
   425→                          formatter={(v: number) => fmt(v)}
   426→                          style={{ fontSize: 10, fill: "var(--muted-foreground)" }}

   428→                      </Bar>
   429→                    </BarChart>
   430→                  </ResponsiveContainer>
   431→                )}

   433→            </CardSection>
   434→
   435→            <CardSection

   437→              description="Discovery → Engagement drop-off"

   445→          </div>

   453→                    <h3 className="text-sm font-semibold">Per-location Breakdown</h3>

   469→                    <div className="text-center py-10 text-sm text-muted-foreground">No location-level data for this selection.</div>
   470→                  ) : (
   471→                    <div className="max-h-96 overflow-y-auto scroll-area rounded-md">
   472→                      <Table>

   482→                          </TableRow>
   483→                        </TableHeader>
   484→                        <TableBody>
   485→                          {tableRows.map((row) => (

   489→                                  <div className="size-6 rounded-md bg-primary/10 text-primary flex items-center justify-center shrink-0">
   490→                                    <Building2 className="size-3.5" />
   491→                                  </div>
   492→                                  <div className="min-w-0">
   493→                                    <div className="truncate max-w-[180px]">{row.name}</div>
   494→                                    <div className="text-[10px] text-muted-foreground">{row.city}</div>
   495→                                  </div>

   555→function Funnel({ totals }: { totals: AnalyticsTotals }) {
   556→  const steps = [
   557→    { label: "Search Views", value: totals.searchViews, key: "searchViews" as MetricKey, icon: Search },
   558→    { label: "Maps Views", value: totals.mapsViews, key: "mapsViews" as MetricKey, icon: Map },
   559→    { label: "Website Clicks", value: totals.websiteClicks, key: "websiteClicks" as MetricKey, icon: MousePointerClick },
   560→    { label: "Phone Calls", value: totals.phoneCalls, key: "phoneCalls" as MetricKey, icon: Phone },
   561→    { label: "Directions", value: totals.directionRequests, key: "directionRequests" as MetricKey, icon: Navigation },
   562→  ];
   563→  const max = steps[0]?.value || 1;
   564→  return (
   565→    <div className="space-y-2.5">
   566→      {steps.map((s, i) => {
   567→        const widthPct = Math.max(2, Math.round((s.value / max) * 100));
   568→        const convPct = i === 0 ? 100 : Math.round((s.value / (steps[i - 1].value || 1)) * 100);
   569→        return (
   570→          <div key={s.key}>
   571→            <div className="flex items-center justify-between text-xs mb-1">
   572→              <span className="font-medium flex items-center gap-1.5">
   573→                <s.icon className="size-3.5" style={{ color: METRIC_COLORS[s.key] }} />
   574→                {s.label}
   575→              </span>
   576→              <span className="tabular-nums text-muted-foreground">
   577→                <span className="text-foreground font-semibold">{fmt(s.value)}</span>
   578→                {i > 0 && <span className="ml-1.5 text-[10px]">({convPct}%)</span>}
   579→              </span>
   580→            </div>
   581→            <div className="h-6 rounded-md bg-muted overflow-hidden">
   582→              <div
   583→                className="h-full rounded-md transition-all flex items-center justify-end pr-2 text-[10px] font-semibold text-white"
   584→                style={{ width: `${widthPct}%`, backgroundColor: METRIC_HEX[s.key] }}
   585→              >
   586→                {widthPct >= 8 ? `${widthPct}%` : ""}
   587→              </div>
   588→            </div>
   589→          </div>
   590→        );
   591→      })}
   592→      <div className="pt-2 mt-2 border-t text-[11px] text-muted-foreground leading-relaxed">
   593→        Overall conversion: <span className="font-semibold text-foreground">
   594→          {totals.searchViews > 0 ? Math.round(((totals.websiteClicks + totals.phoneCalls + totals.directionRequests) / totals.searchViews) * 100) : 0}%
   595→        </span> of discovery actions converted to engagement.
   596→      </div>
   597→    </div>
   598→  );
   599→}
