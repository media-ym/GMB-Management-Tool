    24→export function getGoogleAuthUrl(state?: string): { url: string; state: string } {
    25→  // Generate a high-entropy CSRF state if none provided.
    26→  // The caller is responsible for storing this value (typically in a cookie)
    27→  // and validating it on the OAuth callback.
    28→  const finalState = state || crypto.randomUUID();
    29→  const params = new URLSearchParams({
    30→    client_id: GOOGLE_CLIENT_ID,
    31→    redirect_uri: GOOGLE_REDIRECT_URI,
    32→    response_type: "code",
    33→    scope: GBP_SCOPES,
    34→    access_type: "offline",
    35→    prompt: "consent",
    36→    state: finalState,
    37→  });
    38→  return { url: `https://accounts.google.com/o/oauth2/v2/auth?${params.toString()}`, state: finalState };
    39→}

    41→export async function exchangeCodeForTokens(code: string): Promise<{
    42→  accessToken: string;
    43→  refreshToken: string;
    44→  expiryDate: number;
    45→  scope?: string;
    46→}> {
    47→  const body = new URLSearchParams({
    48→    code,
    49→    client_id: GOOGLE_CLIENT_ID,
    50→    client_secret: GOOGLE_CLIENT_SECRET,
    51→    redirect_uri: GOOGLE_REDIRECT_URI,
    52→    grant_type: "authorization_code",
    53→  });
    54→
    55→  const res = await fetch("https://oauth2.googleapis.com/token", {
    56→    method: "POST",
    57→    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    58→    body,
    59→  });
    60→
    61→  if (!res.ok) {
    62→    const err = await res.text();
    63→    throw new Error(sanitizeGoogleError(`Google token exchange failed: ${res.status} ${err}`));
    64→  }
    65→
    66→  const tokens = await res.json();
    67→  return {
    68→    accessToken: tokens.access_token,
    69→    refreshToken: tokens.refresh_token,
    70→    expiryDate: Date.now() + (tokens.expires_in ?? 3600) * 1000,
    71→    scope: typeof tokens.scope === "string" ? tokens.scope : undefined,
    72→  };
    73→}

    75→export async function refreshAccessToken(refreshToken: string): Promise<{
    76→  accessToken: string;
    77→  expiryDate: number;
    78→}> {
    79→  const body = new URLSearchParams({
    80→    client_id: GOOGLE_CLIENT_ID,
    81→    client_secret: GOOGLE_CLIENT_SECRET,
    82→    refresh_token: refreshToken,
    83→    grant_type: "refresh_token",
    84→  });
    85→
    86→  const res = await fetch("https://oauth2.googleapis.com/token", {
    87→    method: "POST",
    88→    headers: { "Content-Type": "application/x-www-form-urlencoded" },
    89→    body,
    90→  });
    91→
    92→  if (!res.ok) throw new Error(sanitizeGoogleError(`Token refresh failed: ${res.status}`));
    93→  const tokens = await res.json();
    94→  return {
    95→    accessToken: tokens.access_token,
    96→    expiryDate: Date.now() + (tokens.expires_in ?? 3600) * 1000,
    97→  };
    98→}

   100→export async function getValidAccessToken(): Promise<string | null> {
   101→  const account = await db.googleAccount.findFirst();
   102→  if (!account) return null;
   103→
   104→  // Defensive: if the stored token is the legacy "no_token" placeholder, treat as no token.
   105→  if (account.accessToken === "no_token") return null;
   106→
   107→  // Check if token is still valid (5 min buffer)
   108→  if (account.tokenExpiry && new Date(account.tokenExpiry) > new Date(Date.now() + 5 * 60 * 1000)) {
   109→    const decrypted = decryptToken(account.accessToken);
   110→    if (!decrypted) {
   111→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   112→      return null;
   113→    }
   114→    return decrypted;
   115→  }
   116→
   117→  // Try to refresh — decrypt the refresh token first
   118→  if (account.refreshToken) {
   119→    const decryptedRefresh = decryptToken(account.refreshToken);
   120→    if (!decryptedRefresh) {
   121→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   122→      return null;
   123→    }
   124→    try {
   125→      const { accessToken, expiryDate } = await refreshAccessToken(decryptedRefresh);
   126→      await db.googleAccount.update({
   127→        where: { id: account.id },
   128→        data: {
   129→          accessToken: encryptToken(accessToken), // ENCRYPT before saving
   130→          tokenExpiry: new Date(expiryDate),
   131→          status: "active",
   132→        },
   133→      });
   134→      return accessToken; // return decrypted
   135→    } catch (e) {
   136→      console.error("Token refresh failed:", e);
   137→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   138→      return null;
   139→    }
   140→  }
   141→
   142→  return null;
   143→}

   206→/**
   207→ * Search Google's category database by name. Returns matching categories
   208→ * with their gcid IDs (e.g. "gcid:interior_designer"). Used to resolve a
   209→ * human-readable category display name into the stable categoryId required
   210→ * by the Location patch endpoint.
   211→ */
   212→export async function searchGoogleCategories(
   213→  accessToken: string,
   214→  searchTerm: string,
   215→  regionCode: string = "IN",
   216→  languageCode: string = "en"
   217→): Promise<{ categoryId: string; displayName: string }[]> {
   218→  const url = new URL(`${GBP_API_BASE}/categories:search`);
   219→  url.searchParams.set("searchTerm", searchTerm);
   220→  url.searchParams.set("regionCode", regionCode);
   221→  url.searchParams.set("languageCode", languageCode);
   222→
   223→  const data = await withRetry<any>(() =>
   224→    fetch(url.toString(), {
   225→      headers: { Authorization: `Bearer ${accessToken}` },
   226→    }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() }))
   227→  );
   228→  return (data.categories || []).map((c: any) => ({
   229→    categoryId: c.name, // e.g. "gcid:interior_designer"
   230→    displayName: c.displayName,
   231→  }));
   232→}
   233→
   234→/**
   235→ * Resolve a category display name to its stable gcid by searching Google's
   236→ * category DB. Prefers an exact (case-insensitive) display-name match and
   237→ * falls back to the first search hit. Returns null if no match is found so
   238→ * the caller can decide whether to skip that category entirely (Google
   239→ * rejects null/empty categoryId values, so we must not send one).
   240→ */
   241→export async function resolveCategoryId(
   242→  accessToken: string,
   243→  displayName: string,
   244→  regionCode: string = "IN"
   245→): Promise<string | null> {
   246→  const results = await searchGoogleCategories(accessToken, displayName, regionCode);
   247→  // Exact match first (case-insensitive)
   248→  const exact = results.find(r => r.displayName.toLowerCase() === displayName.toLowerCase());
   249→  if (exact) return exact.categoryId;
   250→  // Fall back to first result, if any
   251→  return results[0]?.categoryId ?? null;
   252→}

   343→// ─── Media / Photos (push TO Google + delete FROM Google) ──────────────────
   344→//
   345→// Google's Business Information API supports two upload flows for photos:
   346→//   (1) 2-step byte upload: startUpload → PUT raw bytes to media upload endpoint
   347→//       → POST {parent}/media with the resulting dataRef. Required when the
   348→//       image only exists locally as a binary blob.
   349→//   (2) sourceUrl method (single step): POST {parent}/media with body
   350→//       `{ mediaFormat: { photo: { sourceUrl } }, locationAssociation: { category } }`.
   351→//       Google fetches the image from the public URL itself. Simpler and
   352→//       sufficient when the asset is already hosted at a publicly reachable URL.
   353→//
   354→// We use method (2) — every upload is first persisted to /public/uploads/media
   355→// and exposed via NEXTAUTH_URL, then handed to Google as a sourceUrl. This
   356→// keeps our upload path single-step and avoids the fragile media-upload
   357→// endpoint, while still satisfying Google's requirement that the photo be
   358→// reachable at the time of indexing.
   359→
   360→export type GooglePhotoCategory =
   361→  | "COVER"
   362→  | "PROFILE"
   363→  | "INTERIOR"
   364→  | "EXTERIOR"
   365→  | "PRODUCT"
   366→  | "TEAM"
   367→  | "FOOD_AND_DRINK"
   368→  | "MENU"
   369→  | "AT_WORK"
   370→  | "COMMON_AREA"
   371→  | "ROOMS"

   390→/**
   391→ * Upload a photo to a Google Business Profile location using the sourceUrl
   392→ * method. `locationName` is the Google-side location name like
   393→ * "accounts/{aid}/locations/{lid}". Throws on any non-2xx response (sanitized
   394→ * by withRetry). Callers should wrap in try/catch and treat failures as
   395→ * best-effort — the local MediaLibrary record remains the source of truth.
   396→ */
   397→export async function uploadGooglePhoto(
   398→  accessToken: string,
   399→  locationName: string,
   400→  photo: UploadGooglePhotoInput,
   401→): Promise<UploadGooglePhotoResult> {
   402→  const body: any = {
   403→    mediaFormat: {
   404→      photo: {
   405→        sourceUrl: photo.sourceUrl,
   406→      },
   407→    },
   408→  };
   409→  if (photo.description) {
   410→    body.mediaFormat.photo.description = photo.description;
   411→  }
   412→  if (photo.category) {
   413→    body.locationAssociation = { category: photo.category };
   414→  }
   415→
   416→  const data = await withRetry<any>(() =>
   417→    fetch(`${GBP_API_BASE}/${locationName}/media`, {
   418→      method: "POST",
   419→      headers: {
   420→        Authorization: `Bearer ${accessToken}`,
   421→        "Content-Type": "application/json",
   422→      },
   423→      body: JSON.stringify(body),
   424→    }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() }))
   425→  );
   426→
   427→  return {
   428→    name: data.name,
   429→    googleUrl: data.googleUrl ?? undefined,
   430→  };
   431→}

   433→/**
   434→ * Delete a media item from a Google Business Profile location. `mediaName` is
   435→ * the full Google media name like "accounts/{aid}/locations/{lid}/media/{mid}".
   436→ *
   437→ * A 404 response is treated as success (the photo was already deleted on
   438→ * Google's side — common during sync drift). All other non-2xx responses throw
   439→ * via withRetry. Callers should treat the throw as best-effort and still
   440→ * remove the local record so the operator sees a consistent state.
   441→ */
   442→export async function deleteGooglePhoto(
   443→  accessToken: string,
   444→  mediaName: string,
   445→): Promise<boolean> {
   446→  await withRetry<any>(() =>
   447→    fetch(`${GBP_API_BASE}/${mediaName}`, {
   448→      method: "DELETE",
   449→      headers: { Authorization: `Bearer ${accessToken}` },
   450→    }).then(async (r) => {
   451→      // Idempotent: 404 means the photo is already gone on Google's side.
   452→      // Normalize to a success-shape so withRetry doesn't throw.
   453→      if (r.status === 404) {
   454→        return { ok: true, status: 200, body: () => Promise.resolve("{}") };
   455→      }
   456→      return { ok: r.ok, status: r.status, body: () => r.text() };
   457→    })
   458→  );
   459→  return true;
   460→}

   488→// Fetch ALL 5 metric types for a date range and return as daily aggregates
   489→export async function getFullPerformanceMetrics(
   490→  accessToken: string,
   491→  locationName: string,
   492→  daysBack: number = 30
   493→): Promise<{ date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }[]> {
   494→  const today = new Date();
   495→  const start = new Date(today);
   496→  start.setDate(today.getDate() - daysBack);
   497→
   498→  const startDate = { year: start.getFullYear(), month: start.getMonth() + 1, day: start.getDate() };
   499→  const endDate = { year: today.getFullYear(), month: today.getMonth() + 1, day: today.getDate() };
   500→
   501→  // Google API returns one metric type per call, so fetch all 5 in parallel
   502→  const metricTypes = {
   503→    searchViews: "BUSINESS_IMPRESSIONS_DESKTOP_SEARCH",
   504→    mapsViews: "BUSINESS_IMPRESSIONS_DESKTOP_MAPS",
   505→    websiteClicks: "WEBSITE_CLICKS",
   506→    phoneCalls: "CALL_CLICKS",
   507→    directionRequests: "BUSINESS_DIRECTION_REQUESTS",
   508→  };
   509→
   510→  const [searchRes, mapsRes, websiteRes, callRes, directionRes] = await Promise.allSettled([
   511→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.searchViews),
   512→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.mapsViews),
   513→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.websiteClicks),
   514→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.phoneCalls),
   515→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.directionRequests),
   516→  ]);
   517→
   518→  // Build a map of date → metrics
   519→  const dailyMap = new Map<string, { date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }>();
   520→
   521→  function populate(result: PromiseSettledResult<any[]>, key: keyof typeof metricTypes) {
   522→    if (result.status !== "fulfilled") return;
   523→    for (const entry of result.value) {
   524→      const dateStr = entry.date?.year
   525→        ? `${entry.date.year}-${String(entry.date.month).padStart(2, "0")}-${String(entry.date.day).padStart(2, "0")}`
   526→        : null;
   527→      if (!dateStr) continue;
   528→      if (!dailyMap.has(dateStr)) {
   529→        dailyMap.set(dateStr, {
   530→          date: new Date(dateStr),
   531→          searchViews: 0, mapsViews: 0, websiteClicks: 0, phoneCalls: 0, directionRequests: 0,
   532→        });
   533→      }
   534→      const day = dailyMap.get(dateStr)!;
   535→      const value = entry.value ?? 0;
   536→      (day as any)[key] += value;
   537→    }
   538→  }
   539→
   540→  populate(searchRes, "searchViews");
   541→  populate(mapsRes, "mapsViews");
   542→  populate(websiteRes, "websiteClicks");
   543→  populate(callRes, "phoneCalls");
   544→  populate(directionRes, "directionRequests");
   545→
   546→  return Array.from(dailyMap.values()).sort((a, b) => a.date.getTime() - b.date.getTime());
   547→}

   597→// ─── Update Google Business Profile (push changes TO Google) ───────────────
   598→
   599→export async function updateGoogleBusinessProfile(
   600→  accessToken: string,
   601→  locationName: string,
   602→  updates: {
   603→    title?: string;
   604→    phone?: string;
   605→    website?: string;
   606→    description?: string;
   607→    appointmentUrl?: string;
   608→    hours?: { dayOfWeek: string; openTime?: { hours: number; minutes: number }; closeTime?: { hours: number; minutes: number }; isClosed?: boolean }[];
   609→    /**
   610→     * Categories to push to Google. Callers may supply either the stable
   611→     * gcid (preferred — e.g. "gcid:interior_designer") OR a display name
   612→     * (fallback — will be resolved to a gcid via searchGoogleCategories).
   613→     * Display-name resolution is best-effort: if a name cannot be resolved
   614→     * to a gcid, that category is silently dropped from the patch (Google
   615→     * rejects null/empty categoryId values).
   616→     */
   617→    categories?: {
   618→      primaryCategoryId?: string;
   619→      primaryDisplayName?: string;
   620→      additionalCategoryIds?: string[];
   621→      additionalDisplayNames?: string[];
   622→    };
   623→  }
   624→): Promise<any> {
   625→  const body: any = {};
   626→  const fieldMask: string[] = [];
   627→
   628→  if (updates.title !== undefined) { body.title = updates.title; fieldMask.push("title"); }
   629→  if (updates.phone !== undefined) { body.phoneNumbers = { primaryPhone: updates.phone }; fieldMask.push("phoneNumbers"); }
   630→  if (updates.website !== undefined) { body.websiteUri = updates.website; fieldMask.push("websiteUri"); }
   631→  if (updates.description !== undefined) {
   632→    body.profile = { ...(body.profile || {}), description: updates.description };
   633→    fieldMask.push("profile.description");
   634→  }
   635→  if (updates.appointmentUrl !== undefined) {
   636→    body.profile = { ...(body.profile || {}), appointmentUrl: updates.appointmentUrl };
   637→    fieldMask.push("profile.appointmentUrl");
   638→  }
   639→  if (updates.hours?.length > 0) {
   640→    body.regularHours = {
   641→      periods: updates.hours.map(h => ({
   642→        openDay: h.dayOfWeek,
   643→        openTime: h.openTime,
   644→        closeDay: h.dayOfWeek,
   645→        closeTime: h.closeTime,
   646→      })),
   647→    };
   648→    fieldMask.push("regularHours");
   649→  }
   650→  if (updates.categories) {
   651→    const cat = updates.categories;
   652→    // Resolve primary category — prefer an explicit gcid, else resolve the
   653→    // display name via Google's category search.
   654→    let primaryId = cat.primaryCategoryId;
   655→    if (!primaryId && cat.primaryDisplayName) {
   656→      primaryId = await resolveCategoryId(accessToken, cat.primaryDisplayName);
   657→    }
   658→    // Resolve additional categories — explicit gcids first, then resolve
   659→    // display names. Skip any that can't be resolved (Google rejects null
   660→    // categoryId values, so dropping is safer than sending one).
   661→    const additionalIds: string[] = [];
   662→    if (cat.additionalCategoryIds) {
   663→      additionalIds.push(...cat.additionalCategoryIds.filter(Boolean));
   664→    }
   665→    if (cat.additionalDisplayNames) {
   666→      for (const name of cat.additionalDisplayNames) {
   667→        if (!name) continue;
   668→        const id = await resolveCategoryId(accessToken, name);
   669→        if (id) additionalIds.push(id);
   670→      }
   671→    }
   672→
   673→    body.categories = {
   674→      ...(primaryId ? { primaryCategory: { categoryId: primaryId } } : {}),
   675→      additionalCategories: additionalIds.map(id => ({ categoryId: id })),
   676→    };
   677→    fieldMask.push("categories");
   678→  }
   679→
   680→  const data = await withRetry<any>(() =>
   681→    fetch(`${GBP_API_BASE}/${locationName}?updateMask=${fieldMask.join(",")}&validateOnly=false`, {
   682→      method: "PATCH",
   683→      headers: {
   684→        Authorization: `Bearer ${accessToken}`,
   685→        "Content-Type": "application/json",
   686→      },
   687→      body: JSON.stringify(body),
   688→    }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() }))
   689→  );
   690→  return data;
   691→}

   693→// ─── Google Business Profile Verifications API ─────────────────────────────
   694→// Uses a separate base URL from the rest of the GBP APIs. See:
   695→// https://developers.google.com/my-business/reference/verifications/v1/rest
   696→const GBP_VERIFY_BASE = "https://mybusinessverifications.googleapis.com/v1";
   697→
   698→/**
   699→ * Fetch the verification options available for a location.
   700→ * The optional `dispatchMethod` lets the caller pre-select the channel they
   701→ * intend to use (ADDRESS | EMAIL | PHONE_CALL | SMS) so Google can return
   702→ * channel-specific metadata (e.g. the masked phone number for SMS).
   703→ */
   704→export async function fetchVerificationOptions(
   705→  accessToken: string,
   706→  locationName: string,
   707→  dispatchMethod?: string,
   708→): Promise<any> {
   709→  const body: any = {};
   710→  if (dispatchMethod) {
   711→    body.context = { dispatchMethod };
   712→  }
   713→  const data = await withRetry<any>(() =>
   714→    fetch(`${GBP_VERIFY_BASE}/${locationName}:fetchVerificationOptions`, {
   715→      method: "POST",
   716→      headers: { Authorization: `Bearer ${accessToken}`, "Content-Type": "application/json" },
   717→      body: JSON.stringify(body),
   718→    }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() })),
   719→  );
   720→  return data;
   721→}
   722→
   723→/**
   724→ * Initiate a verification flow for a location. Google will dispatch the PIN
   725→ * (postcard for ADDRESS, SMS for SMS, automated call for PHONE_CALL, email
   726→ * for EMAIL) and return a `verification` resource the client must later
   727→ * complete via {@link completeVerification}.
   728→ *
   729→ * `input` shape depends on `method`:
   730→ *   - ADDRESS → { mailerContactName: string }
   731→ *   - PHONE_CALL | SMS → { phoneNumber: string }
   732→ *   - EMAIL → { emailAddress: string }
   733→ */
   734→export async function initiateVerification(
   735→  accessToken: string,
   736→  locationName: string,
   737→  method: string,
   738→  input: any,
   739→): Promise<any> {
   740→  const body: any = { method };
   741→  if (method === "ADDRESS") {
   742→    body.addressInput = input;
   743→  } else if (method === "PHONE_CALL" || method === "SMS") {
   744→    body.phoneInput = input;
   745→  } else if (method === "EMAIL") {
   746→    body.emailInput = input;
   747→  }
   748→  const data = await withRetry<any>(() =>
   749→    fetch(`${GBP_VERIFY_BASE}/${locationName}:verify`, {
   750→      method: "POST",
   751→      headers: { Authorization: `Bearer ${accessToken}`, "Content-Type": "application/json" },
   752→      body: JSON.stringify(body),
   753→    }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() })),
   754→  );
   755→  return data;
   756→}

   758→/**
   759→ * List the verification history for a location. Paginates through all
   760→ * available pages (capped at 5 pages = 250 records) and returns the merged
   761→ * `verifications` array.
   762→ */
   763→export async function listVerifications(
   764→  accessToken: string,
   765→  locationName: string,
   766→  pageSize = 50,
   767→): Promise<any[]> {
   768→  const all: any[] = [];
   769→  let pageToken: string | undefined;
   770→  for (let page = 0; page < 5; page++) {
   771→    const url = new URL(`${GBP_VERIFY_BASE}/${locationName}/verifications`);
   772→    url.searchParams.set("pageSize", String(pageSize));
   773→    if (pageToken) url.searchParams.set("pageToken", pageToken);
   774→    const data = await withRetry<any>(() =>
   775→      fetch(url.toString(), {
   776→        headers: { Authorization: `Bearer ${accessToken}` },
   777→      }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() })),
   778→    );
   779→    if (Array.isArray(data.verifications)) all.push(...data.verifications);
   780→    pageToken = data.nextPageToken;
   781→    if (!pageToken) break;
   782→  }
   783→  return all;
   784→}

   786→/**
   787→ * Complete a PIN-based verification by submitting the PIN the user received
   788→ * (postcard in the mail, SMS code, etc.). Returns true on success.
   789→ * A 404 response is treated as success — the verification may have already
   790→ * been completed or expired, and the user-facing outcome is the same.
   791→ */
   792→export async function completeVerification(
   793→  accessToken: string,
   794→  verificationName: string,
   795→  pin: string,
   796→): Promise<boolean> {
   797→  await withRetry<any>(() =>
   798→    fetch(`${GBP_VERIFY_BASE}/${verificationName}:complete`, {
   799→      method: "POST",
   800→      headers: { Authorization: `Bearer ${accessToken}`, "Content-Type": "application/json" },
   801→      body: JSON.stringify({ pin }),
   802→    }).then(async (r) => {
   803→      if (r.status === 404) return { ok: true, status: 200, body: () => "{}" };
   804→      return { ok: r.ok, status: r.status, body: () => r.text() };
   805→    }),
   806→  );
   807→  return true;
   808→}

  1032→              rating: review.starRating === "FIVE" ? 5 : review.starRating === "FOUR" ? 4 : review.starRating === "THREE" ? 3 : review.starRating === "TWO" ? 2 : 1,
  1033→              text: review.comment || "",
  1034→              sentiment: "neutral",
  1035→              replyText: review.reviewReply?.comment || null,
  1036→              replySource: review.reviewReply ? "manual" : null,
  1037→              replyStatus: review.reviewReply ? "replied" : "pending",
  1038→              repliedAt: review.reviewReply ? new Date(review.reviewReply.updateTime) : null,
  1039→              createdAt: new Date(review.createTime),
  1040→            },
  1041→          });
  1042→          result.reviews++;
  1043→        }
  1044→      }
  1045→    } catch (e: any) {
  1046→      errors.push(`Reviews sync: ${e.message}`);
  1047→    }
  1048→
  1049→    // ─── 6. Sync Photos (metadata only — URLs from Google) ──────────────
  1050→    try {
  1051→      const mediaData = await withRetry<{ media?: any[] }>(() =>
  1052→        fetch(`${GBP_API_BASE}/${locationName}/media`, {
  1053→          headers: { Authorization: `Bearer ${accessToken}` },
  1054→        }).then(async (r) => ({ ok: r.ok, status: r.status, body: () => r.text() }))
  1055→      );
  1056→      for (const photo of mediaData.media || []) {
  1057→        const existing = await db.businessPhoto.findFirst({ where: { googlePhotoId: photo.name } });
  1058→        if (!existing) {
  1059→          await db.businessPhoto.create({
  1060→            data: {
  1061→              locationId,
  1062→              googlePhotoId: photo.name,
  1063→              imageUrl: photo.googleUrl || photo.locationUri || "",
  1064→              thumbnailUrl: null,
  1065→              source: "google",
  1066→              status: "active",
  1067→            },
  1068→          });
  1069→          result.photos++;
  1070→        }
  1071→      }
  1072→    } catch (e: any) {
  1073→      errors.push(`Photos sync: ${e.message}`);
  1074→    }
  1075→
  1076→    // ─── 7. Sync Business Information (description, website, appointment URL) ──
  1077→    try {
  1078→      const description = profile.profile?.description || profile.metadata?.description || null;
  1079→      const website = profile.websiteUri || null;
  1080→      const appointmentUrl = profile.profile?.appointmentUrl || null;
  1081→
  1082→      if (gbp) {
  1083→        const existingInfo = await db.businessInformation.findFirst({ where: { profileId: gbp.id } });
  1084→        if (existingInfo) {
  1085→          await db.businessInformation.update({
  1086→            where: { id: existingInfo.id },
  1087→            data: { description, website, appointmentUrl },
  1088→          });
  1089→        } else {
  1090→          await db.businessInformation.create({
  1091→            data: { profileId: gbp.id, locationId, description, website, appointmentUrl },
  1092→          });
  1093→        }
  1094→      }
  1095→    } catch (e: any) {
  1096→      errors.push(`Business info sync: ${e.message}`);
  1097→    }
  1098→
  1099→    // ─── 8. Sync Attributes (wheelchair accessible, appointments, etc.) ────
  1100→    try {
  1101→      if (profile.attributes?.length > 0) {
  1102→        await db.businessAttribute.deleteMany({ where: { locationId } });
  1103→        for (const attr of profile.attributes) {
  1104→          await db.businessAttribute.create({
  1105→            data: {
  1106→              locationId,
  1107→              attributeName: attr.name || attr.displayName || "Attribute",
  1108→              attributeValue: Array.isArray(attr.value) ? attr.value.join(", ") : String(attr.value ?? "true"),
  1109→            },
  1110→          });
  1111→        }
  1112→      }
  1113→    } catch (e: any) {
  1114→      errors.push(`Attributes sync: ${e.message}`);
  1115→    }
  1116→
  1117→    // ─── 9. Sync Special Hours (holidays) ─────────────────────────────────
  1118→    try {
  1119→      if (profile.specialHours?.specialHourRanges?.length > 0) {
  1120→        await db.specialHour.deleteMany({ where: { locationId } });
  1121→        for (const range of profile.specialHours.specialHourRanges) {
  1122→          const startDate = range.startDate ? new Date(`${range.startDate.year}-${String(range.startDate.month).padStart(2, "0")}-${String(range.startDate.day).padStart(2, "0")}`) : new Date();
  1123→          await db.specialHour.create({
  1124→            data: {
  1125→              locationId,
  1126→              date: startDate,
  1127→              openTime: range.openTime ? `${String(range.openTime.hours).padStart(2, "0")}:${String(range.openTime.minutes).padStart(2, "0")}` : null,
  1128→              closeTime: range.closeTime ? `${String(range.closeTime.hours).padStart(2, "0")}:${String(range.closeTime.minutes).padStart(2, "0")}` : null,
  1129→              isClosed: !range.openTime,
  1130→            },
  1131→          });
  1132→        }
  1133→      }
  1134→    } catch (e: any) {
  1135→      errors.push(`Special hours sync: ${e.message}`);
  1136→    }
  1137→
  1138→    // ─── 10. Sync Analytics (real Google Business Performance API) ────────
  1139→    try {
  1140→      const analyticsResult = await syncLocationAnalytics(locationId, 30);
  1141→      if (analyticsResult.errors.length > 0) {
  1142→        errors.push(`Analytics sync: ${analyticsResult.errors.join("; ")}`);
  1143→      }
  1144→    } catch (e: any) {
  1145→      errors.push(`Analytics sync: ${e.message}`);
  1146→    }
  1147→
  1148→    return { success: true, synced: result, errors };
  1149→  } catch (e: any) {
  1150→    return { success: false, synced: result, errors: [e.message] };
  1151→  }
  1152→}
