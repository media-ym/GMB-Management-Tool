    64→    64→function rankColor(rank: number): { bg: string; text: string; label: string } {
    65→    65→  if (!rank || rank <= 0) return { bg: "bg-slate-300 dark:bg-slate-700", text: "text-slate-600 dark:text-slate-300", label: "Not ranked" };
    66→    66→  if (rank <= 3) return { bg: "bg-emerald-500", text: "text-white", label: "Top 3" };
    67→    67→  if (rank <= 10) return { bg: "bg-amber-500", text: "text-white", label: "Top 10" };
    68→    68→  if (rank <= 20) return { bg: "bg-orange-500", text: "text-white", label: "Top 20" };
    69→    69→  return { bg: "bg-rose-500", text: "text-white", label: "21+" };
    70→    70→}

    88→    88→// Deterministic mock trend from a keyword string (seed data has no trend series)
    89→    89→function mockTrend(keyword: string): { delta: number; dir: "up" | "down" | "flat" } {
    90→    90→  let h = 0;
    91→    91→  for (let i = 0; i < keyword.length; i++) h = (h * 31 + keyword.charCodeAt(i)) >>> 0;
    92→    92→  const v = (h % 7) - 3; // -3..+3
    93→    93→  return { delta: Math.abs(v), dir: v > 0 ? "up" : v < 0 ? "down" : "flat" };
    94→    94→}

    96→    96→// ── Build a normalized 5x5 display grid from a flat list of GeoGridPoints ─
    97→    97→interface GridCell {
    98→    98→  lat: number;
    99→    99→  lng: number;
   100→   100→  rank: number;
   101→   101→  row: number; // 0 = north (top)
   102→   102→  col: number; // 0 = west (left)
   103→   103→}
   104→   104→
   105→   105→function buildGrid(points: GeoGridPoint[]): GridCell[] {
   106→   106→  if (!points || points.length === 0) return [];
   107→   107→  const lats = Array.from(new Set(points.map((p) => p.lat))).sort((a, b) => b - a); // desc (N→S)
   108→   108→  const lngs = Array.from(new Set(points.map((p) => p.lng))).sort((a, b) => a - b); // asc (W→E)
   109→   109→  return points.map((p) => ({
   110→   110→    lat: p.lat,
   111→   111→    lng: p.lng,
   112→   112→    rank: p.rank,
   113→   113→    row: lats.indexOf(p.lat),
   114→   114→    col: lngs.indexOf(p.lng),
   115→   115→  }));
   116→   116→}

   416→   416→// ── Geo-grid heatmap (centerpiece) ───────────────────────────────────
   417→   417→function GeoGrid({ keyword, locationName }: { keyword: SeoKeyword; locationName: string }) {
   418→   418→  const cells = useMemo(() => buildGrid(keyword.grid), [keyword.grid]);
   419→   419→
   420→   420→  if (cells.length === 0) {
   421→   421→    return (
   422→   422→      <div className="py-12 text-center text-sm text-muted-foreground">
   423→   423→        No geo-grid data available for this keyword.
   424→   424→      </div>
   425→   425→    );
   426→   426→  }
   427→   427→
   428→   428→  // Determine grid size (assume square)
   429→   429→  const size = Math.max(
   430→   430→    ...cells.map((c) => c.row + 1),
   431→   431→    ...cells.map((c) => c.col + 1),
   432→   432→  );
   433→   433→
   434→   434→  // Build 2D array [row][col]
   435→   435→  const grid: (GridCell | null)[][] = Array.from({ length: size }, () =>
   436→   436→    Array.from({ length: size }, () => null),
   437→   437→  );
   438→   438→  cells.forEach((c) => {
   439→   439→    grid[c.row][c.col] = c;
   440→   440→  });
   441→   441→
   442→   442→  // Compute summary stats
   443→   443→  const ranked = cells.filter((c) => c.rank > 0);
   444→   444→  const top3 = ranked.filter((c) => c.rank <= 3).length;
   445→   445→  const top10 = ranked.filter((c) => c.rank <= 10).length;
   446→   446→  const avg = ranked.length
   447→   447→    ? Math.round((ranked.reduce((s, c) => s + c.rank, 0) / ranked.length) * 10) / 10
   448→   448→    : 0;
   449→   449→
   450→   450→  return (
   451→   451→    <div className="space-y-4">

   594→   594→function KeywordTable({
   595→   595→  keywords, isLoading, selectedKeywordId, onSelect,
   596→   596→}: {
   597→   597→  keywords: SeoKeyword[];
   598→   598→  isLoading: boolean;
   599→   599→  selectedKeywordId: string | null;
   600→   600→  onSelect: (id: string) => void;
   601→   601→}) {
   602→   602→  const [sortKey, setSortKey] = useState<SortKey>("avgRank");
   603→   603→  const [sortDir, setSortDir] = useState<SortDir>("asc");
   604→   604→
   605→   605→  const sorted = useMemo(() => {
   606→   606→    const arr = [...keywords];
   607→   607→    arr.sort((a, b) => {
   608→   608→      let av: number | string = 0;
   609→   609→      let bv: number | string = 0;
   610→   610→      switch (sortKey) {
   611→   611→        case "keyword": av = a.keyword.toLowerCase(); bv = b.keyword.toLowerCase(); break;
   612→   612→        case "city": av = a.city.toLowerCase(); bv = b.city.toLowerCase(); break;
   613→   613→        case "avgRank": av = a.avgRank || 99; bv = b.avgRank || 99; break;
   614→   614→        case "topRank": av = a.topRank ?? 99; bv = b.topRank ?? 99; break;
   615→   615→      }
   616→   616→      if (av < bv) return sortDir === "asc" ? -1 : 1;
   617→   617→      if (av > bv) return sortDir === "asc" ? 1 : -1;
   618→   618→      return 0;
   619→   619→    });
   620→   620→    return arr;
   621→   621→  }, [keywords, sortKey, sortDir]);
   622→   622→

   624→   624→    if (sortKey === key) {
   625→   625→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   626→   626→    } else {
   627→   627→      setSortKey(key);
   628→   628→      setSortDir(key === "keyword" || key === "city" ? "asc" : "asc");
   629→   629→    }
   630→   630→  }
   631→   631→
   632→   632→  if (isLoading) {
   633→   633→    return <Skeleton className="h-64 w-full" />;

   648→   648→          <TableRow className="bg-muted/30">

   651→   651→            <SortableHead label="Avg Rank" k="avgRank" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} align="right" />
   652→   652→            <SortableHead label="Top Rank" k="topRank" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} align="right" />

   655→   655→          </TableRow>
   656→   656→        </TableHeader>
   657→   657→        <TableBody>
   658→   658→          {sorted.map((k) => {
   659→   659→            const isSelected = k.id === selectedKeywordId;
   660→   660→            const trend = mockTrend(k.keyword);
   661→   661→            return (
   662→   662→              <TableRow

   664→   664→                onClick={() => onSelect(k.id)}
   665→   665→                className={cn(
   666→   666→                  "cursor-pointer",

   889→   889→function CompetitorMonitoring({ selectedKeyword }: { selectedKeyword: SeoKeyword | null }) {
   890→   890→  const myRank = selectedKeyword?.avgRank ?? 0;
   891→   891→  const competitors = useMemo(() => {

   896→   896→    return base.sort((a, b) => a.avgRank - b.avgRank);
   897→   897→  }, [myRank]);
   898→   898→

   914→   914→          return (

   916→   916→              <div className="flex items-center justify-between text-xs">

   930→   930→                        "inline-flex items-center text-[10px] tabular-nums",

   932→   932→                      )}
