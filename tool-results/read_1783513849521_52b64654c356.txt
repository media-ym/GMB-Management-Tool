   241→function relativeTime(iso: string | null): string {
   242→  if (!iso) return "Never";
   243→  try { return formatDistanceToNow(new Date(iso), { addSuffix: true }); }
   244→  catch { return "—"; }
   245→}
   246→
   247→function fullTime(iso: string | null): string {
   248→  if (!iso) return "—";
   249→  try { return format(new Date(iso), "d MMM yyyy, h:mm a"); }
   250→  catch { return "—"; }
   251→}

   253→function scoreColor(score: number): string {
   254→  if (score >= 75) return "#10b981";
   255→  if (score >= 50) return "#f59e0b";
   256→  return "#f43f5e";
   257→}
   258→
   259→function scoreColorClass(score: number): string {
   260→  if (score >= 75) return "text-emerald-600 dark:text-emerald-400";
   261→  if (score >= 50) return "text-amber-600 dark:text-amber-400";
   262→  return "text-rose-600 dark:text-rose-400";
   263→}
   264→
   265→function progressIndicatorClass(score: number): string {
   266→  if (score >= 75) return "[&>[data-slot=progress-indicator]]:bg-emerald-500";
   267→  if (score >= 50) return "[&>[data-slot=progress-indicator]]:bg-amber-500";
   268→  return "[&>[data-slot=progress-indicator]]:bg-rose-500";
   269→}
   270→
   271→function currencySymbol(code: string): string {
   272→  return code === "INR" ? "₹" : code === "USD" ? "$" : code === "EUR" ? "€" : code + " ";
   273→}
   274→
   275→function formatPrice(price: number | null, currency: string): string {
   276→  if (price == null) return "—";
   277→  try {
   278→    return currencySymbol(currency) + price.toLocaleString("en-IN", { maximumFractionDigits: 2 });
   279→  } catch {
   280→    return String(price);
   281→  }
   282→}
   283→
   284→function formatAnalytics(n: number | null | undefined): string {
   285→  if (n == null) return "0";
   286→  return n.toLocaleString("en-IN");
   287→}

   293→export function LocationsView() {
   294→  const user = useUser();
   295→  const setView = useAppStore((s) => s.setView);
   296→  const setActiveLocationId = useAppStore((s) => s.setActiveLocationId);
   297→  const qc = useQueryClient();
   298→
   299→  const canSync = can(user.role, "system.sync");
   300→  const canManage = can(user.role, "locations.manage");
   301→
   302→  const [search, setSearch] = useState("");
   303→  const [statusFilter, setStatusFilter] = useState<"all" | LocationStatus>("all");
   304→  const [sort, setSort] = useState<"city" | "rating" | "health" | "reviews">("city");
   305→  const [detailId, setDetailId] = useState<string | null>(null);
   306→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   307→  const [syncingId, setSyncingId] = useState<string | null>(null);
   308→  const [bulkRunning, setBulkRunning] = useState<"sync" | "archive" | "activate" | null>(null);
   309→  const [addOpen, setAddOpen] = useState(false);
   310→
   311→  const { data: locations, isLoading } = useQuery<LocationWithStats[]>({
   312→    queryKey: ["locations"],
   313→    queryFn: () => api<LocationWithStats[]>("/api/locations"),
   314→  });
   315→
   316→  const { data: summary } = useQuery<DashboardSummary>({
   317→    queryKey: ["dashboard-summary"],
   318→    queryFn: () => api<DashboardSummary>("/api/dashboard"),
   319→  });
   320→
   321→  // Reset detail when locations list refetches & detail no longer exists
   322→  useEffect(() => {
   323→    if (detailId && locations && !locations.some((l) => l.id === detailId)) {
   324→      setDetailId(null);
   325→    }
   326→  }, [detailId, locations]);
   327→
   328→  const filtered = useMemo(() => {
   329→    if (!locations) return [];
   330→    const q = search.trim().toLowerCase();
   331→    let list = locations.filter((l) => {
   332→      if (statusFilter !== "all" && l.status !== statusFilter) return false;
   333→      if (!q) return true;
   334→      return (
   335→        l.name.toLowerCase().includes(q) ||
   336→        l.city.toLowerCase().includes(q) ||
   337→        l.region.toLowerCase().includes(q)
   338→      );
   339→    });
   340→    list = list.slice().sort((a, b) => {
   341→      switch (sort) {
   342→        case "rating": return b.avgRating - a.avgRating;
   343→        case "health": return b.healthScore - a.healthScore;
   344→        case "reviews": return b.reviewCount - a.reviewCount;
   345→        case "city":
   346→        default: return a.city.localeCompare(b.city) || a.name.localeCompare(b.name);
   347→      }
   348→    });
   349→    return list;
   350→  }, [locations, search, statusFilter, sort]);
   351→
   352→  const avgHealth = useMemo(() => {
   353→    if (!locations || locations.length === 0) return 0;
   354→    return Math.round(locations.reduce((a, l) => a + l.healthScore, 0) / locations.length);
   355→  }, [locations]);
   356→
   357→  const syncErrors = useMemo(
   358→    () => (locations ?? []).filter((l) => l.syncStatus === "error").length,
   359→    [locations],
   360→  );
   361→
   362→  const activeCount = useMemo(
   363→    () => (locations ?? []).filter((l) => l.status === "active").length,
   364→    [locations],
   365→  );
   366→
   367→  async function handleSyncAll() {
   368→    try {
   369→      toast.loading("Triggering Google sync for all locations…", { id: "sync-all" });
   370→      await api("/api/dashboard", { method: "POST", body: JSON.stringify({}) });
   371→      await qc.invalidateQueries({ queryKey: ["locations"] });
   372→      await qc.invalidateQueries({ queryKey: ["dashboard-summary"] });
   373→      toast.success("Sync complete — all locations updated.", { id: "sync-all" });
   374→    } catch (e: unknown) {
   375→      const msg = e instanceof Error ? e.message : "Sync failed";
   376→      toast.error(msg, { id: "sync-all" });
   377→    }
   378→  }
   379→
   380→  async function handleSyncOne(loc: LocationWithStats, module: SyncModule = "full") {
   381→    const toastId = `sync-${loc.id}-${module}`;
   382→    try {
   383→      setSyncingId(loc.id);
   384→      toast.loading(`Syncing ${module === "full" ? "everything" : module} for ${loc.name}…`, { id: toastId });
   385→      await api(`/api/locations/${loc.id}/sync`, {
   386→        method: "POST",
   387→        body: JSON.stringify({ module }),
   388→      });
   389→      await qc.invalidateQueries({ queryKey: ["locations"] });
   390→      await qc.invalidateQueries({ queryKey: ["dashboard-summary"] });
   391→      if (detailId === loc.id) await qc.invalidateQueries({ queryKey: ["location-detail", loc.id] });
   392→      toast.success(`${loc.name} ${module === "full" ? "synced" : `${module} synced`}.`, { id: toastId });
   393→    } catch (e: unknown) {
   394→      const msg = e instanceof Error ? e.message : "Sync failed";
   395→      toast.error(msg, { id: toastId });
   396→    } finally {
   397→      setSyncingId(null);
   398→    }
   399→  }
   400→
   401→  function openReviews(loc: LocationWithStats) {
   402→    setActiveLocationId(loc.id);
   403→    setView("reviews");
   404→  }
   405→
   406→  function toggleSelected(id: string) {
   407→    setSelectedIds((prev) => {
   408→      const next = new Set(prev);
   409→      if (next.has(id)) next.delete(id);
   410→      else next.add(id);
   411→      return next;
   412→    });
   413→  }
   414→
   415→  function toggleSelectAll() {
   416→    if (!filtered) return;
   417→    if (selectedIds.size === filtered.length) setSelectedIds(new Set());
   418→    else setSelectedIds(new Set(filtered.map((l) => l.id)));
   419→  }
   420→
   421→  function clearSelection() {
   422→    setSelectedIds(new Set());
   423→  }
   424→
   425→  async function handleBulk(action: "sync" | "archive" | "activate") {
   426→    if (selectedIds.size === 0) return;
   427→    const ids = Array.from(selectedIds);
   428→    const labels = {
   429→      sync: "Syncing",
   430→      archive: "Archiving",
   431→      activate: "Activating",
   432→    } as const;
   433→    try {
   434→      setBulkRunning(action);
   435→      toast.loading(`${labels[action]} ${ids.length} location${ids.length === 1 ? "" : "s"}…`, { id: "bulk-locations" });
   436→      await api("/api/locations/bulk", {
   437→        method: "POST",
   438→        body: JSON.stringify({ action, locationIds: ids }),
   439→      });
   440→      await qc.invalidateQueries({ queryKey: ["locations"] });
   441→      await qc.invalidateQueries({ queryKey: ["dashboard-summary"] });
   442→      toast.success(`Done — ${ids.length} location${ids.length === 1 ? "" : "s"} ${action === "sync" ? "synced" : action === "archive" ? "archived" : "activated"}.`, { id: "bulk-locations" });
   443→      clearSelection();
   444→    } catch (e: unknown) {
   445→      const msg = e instanceof Error ? e.message : "Bulk action failed";
   446→      toast.error(msg, { id: "bulk-locations" });
   447→    } finally {
   448→      setBulkRunning(null);
   449→    }
   450→  }
   451→
   452→  // ─── Detail view ──────────────────────────────────────────────────────────
   453→  if (detailId) {
   454→    const loc = (locations ?? []).find((l) => l.id === detailId) ?? null;
   455→    return (
   456→      <LocationDetail
   457→        locationId={detailId}
   458→        fallbackLoc={loc}
   459→        canSync={canSync}
   460→        canManage={canManage}
   461→        syncing={syncingId === detailId}
   462→        onBack={() => setDetailId(null)}
   463→        onSync={(module) => loc ? handleSyncOne(loc, module) : undefined}
   464→        onViewReviews={() => loc ? openReviews(loc) : undefined}
   465→      />
   466→    );
   467→  }
   468→
   469→  // ─── Grid view ────────────────────────────────────────────────────────────
   470→  return (
   471→    <div className="p-4 sm:p-6 lg:p-8 space-y-6">
   472→      <PageHeader
   473→        title="Locations"
   474→        description="Manage all MyFNG Google Business Profiles across cities"
   475→        icon={MapPin}
   476→        actions={
   477→          <>
   478→            {canManage && (
   479→              <Button size="sm" onClick={() => setAddOpen(true)}>
   480→                <Plus className="size-3.5 mr-1.5" /> Add Location
   481→              </Button>
   482→            )}
   483→            {canSync ? (
   484→              <Button size="sm" variant="outline" onClick={handleSyncAll} disabled={isLoading}>
   485→                <RefreshCw className="size-3.5 mr-1.5" /> Sync all
   486→              </Button>
   487→            ) : null}
   488→          </>
   489→        }
   490→      />
   491→
   492→      {/* Summary stat row */}
   493→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 sm:gap-4">
   494→        {isLoading || !locations ? (
   495→          Array.from({ length: 4 }).map((_, i) => (
   496→            <Skeleton key={i} className="h-28 rounded-xl" />
   497→          ))
   498→        ) : (
   499→          <>
   500→            <StatCard
   501→              label="Total Locations"
   502→              value={locations.length}
   503→              icon={Building2}
   504→              hint={`${activeCount} active`}
   505→              accent="emerald"
   506→            />
   507→            <StatCard
   508→              label="Active"
   509→              value={activeCount}
   510→              icon={Activity}
   511→              hint={`${locations.length - activeCount} paused / error`}
   512→              accent="teal"
   513→            />
   514→            <StatCard
   515→              label="Avg Health Score"
   516→              value={avgHealth}
   517→              icon={HeartPulse}
   518→              hint="Across all locations"
   519→              accent="amber"
   520→            />
   521→            <StatCard
   522→              label="Sync Errors"
   523→              value={syncErrors}
   524→              icon={AlertTriangle}
   525→              hint={syncErrors === 0 ? "All healthy" : "Needs attention"}
   526→              accent={syncErrors > 0 ? "rose" : "emerald"}
   527→            />
   528→          </>
   529→        )}
   530→      </div>
   531→
   532→      {/* Bulk action bar */}
   533→      {selectedIds.size > 0 && (
   534→        <Card className="border-primary/30 bg-primary/[0.03]">
   535→          <CardContent className="p-3 flex flex-wrap items-center gap-2 sm:gap-3">
   536→            <div className="flex items-center gap-2 text-sm font-medium">
   537→              <span className="inline-flex items-center justify-center size-6 rounded-full bg-primary text-primary-foreground text-xs font-semibold tabular-nums">
   538→                {selectedIds.size}
   539→              </span>
