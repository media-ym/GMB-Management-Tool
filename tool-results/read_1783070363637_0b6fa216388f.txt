   179→function UsersTab() {
   180→  const qc = useQueryClient();
   181→  const { data: locations } = useLocations();
   182→  const [search, setSearch] = useState("");
   183→  const [dialogOpen, setDialogOpen] = useState(false);
   184→  const [editing, setEditing] = useState<UserRow | null>(null);
   185→  const [togglingId, setTogglingId] = useState<string | null>(null);
   186→
   187→  const { data: users, isLoading } = useQuery<UserRow[]>({
   188→    queryKey: ["users"],
   189→    queryFn: () => api<UserRow[]>("/api/users"),
   190→  });
   191→
   192→  const locationMap = useMemo(() => {
   193→    const m = new Map<string, string>();
   194→    (locations ?? []).forEach((l) => m.set(l.id, l.city || l.name));
   195→    return m;
   196→  }, [locations]);
   197→
   198→  const filtered = useMemo(() => {
   199→    if (!users) return [];
   200→    const q = search.trim().toLowerCase();
   201→    if (!q) return users;
   202→    return users.filter((u) =>
   203→      u.name.toLowerCase().includes(q) ||
   204→      u.email.toLowerCase().includes(q) ||
   205→      roleLabel(u.role).toLowerCase().includes(q),
   206→    );
   207→  }, [users, search]);
   208→
   209→  function openCreate() {
   210→    setEditing(null);
   211→    setDialogOpen(true);
   212→  }
   213→
   214→  function openEdit(u: UserRow) {
   215→    setEditing(u);
   216→    setDialogOpen(true);
   217→  }
   218→
   219→  async function toggleStatus(u: UserRow) {
   220→    setTogglingId(u.id);
   221→    try {
   222→      const newStatus = u.status === "active" ? "inactive" : "active";
   223→      await api("/api/users", {
   224→        method: "PATCH",
   225→        body: JSON.stringify({ id: u.id, status: newStatus }),
   226→      });
   227→      qc.invalidateQueries({ queryKey: ["users"] });
   228→      toast.success(`${u.name} ${newStatus === "active" ? "activated" : "deactivated"}`);
   229→    } catch (e: any) {
   230→      toast.error(e?.message || "Failed to update user");
   231→    } finally {
   232→      setTogglingId(null);
   233→    }
   234→  }
   235→
   236→  async function unlockUser(u: UserRow) {
   237→    setTogglingId(u.id);
   238→    try {
   239→      await api("/api/users", {
   240→        method: "PATCH",
   241→        body: JSON.stringify({ id: u.id, status: "active" }),
   242→      });
   243→      qc.invalidateQueries({ queryKey: ["users"] });
   244→      toast.success(`${u.name} unlocked`);
   245→    } catch (e: any) {
   246→      toast.error(e?.message || "Failed to unlock user");
   247→    } finally {
   248→      setTogglingId(null);
   249→    }
   250→  }
   251→
   252→  return (
   253→    <div className="space-y-4">
   254→      {/* Toolbar */}
   255→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   256→        <div className="relative w-full sm:max-w-xs">
   257→          <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 size-4 text-muted-foreground" />
   258→          <Input
   259→            value={search}
   260→            onChange={(e) => setSearch(e.target.value)}
   261→            placeholder="Search users by name, email, role…"
   262→            className="pl-9 h-9"
   263→          />
   264→        </div>
   265→        <Button size="sm" onClick={openCreate}>
   266→          <UserPlus className="size-3.5 mr-1.5" /> Invite user
   267→        </Button>
   268→      </div>
   269→
   270→      {/* Users table — desktop */}
   271→      <Card className="overflow-hidden hidden md:block">
   272→        {isLoading ? (
   273→          <UsersTableSkeleton />
   274→        ) : filtered.length === 0 ? (
   275→          <UsersEmpty search={search} />
   276→        ) : (
   277→          <Table>
   278→            <TableHeader>
   279→              <TableRow className="bg-muted/40">
   280→                <TableHead className="pl-5">Name</TableHead>
   281→                <TableHead>Email</TableHead>
   282→                <TableHead>Role</TableHead>
   283→                <TableHead>Assigned Locations</TableHead>
   284→                <TableHead>Status</TableHead>
   285→                <TableHead>Last login</TableHead>
   286→                <TableHead>Created</TableHead>
   287→                <TableHead className="pr-5 text-right">Actions</TableHead>
   288→              </TableRow>
   289→            </TableHeader>
   290→            <TableBody>
   291→              {filtered.map((u) => (
   292→                <UserTableRow
   293→                  key={u.id}
   294→                  u={u}
   295→                  locationMap={locationMap}
   296→                  toggling={togglingId === u.id}
   297→                  onToggle={() => toggleStatus(u)}
   298→                  onEdit={() => openEdit(u)}
   299→                />
   300→              ))}
   301→            </TableBody>
   302→          </Table>
   303→        )}
   304→      </Card>
   305→
   306→      {/* Users list — mobile cards */}
   307→      <div className="md:hidden space-y-3">
   308→        {isLoading ? (
   309→          <div className="space-y-3">
   310→            {Array.from({ length: 4 }).map((_, i) => (
   311→              <Skeleton key={i} className="h-32 rounded-xl" />
   312→            ))}
   313→          </div>
   314→        ) : filtered.length === 0 ? (
   315→          <Card><CardContent className="py-0"><UsersEmpty search={search} /></CardContent></Card>
   316→        ) : (
   317→          filtered.map((u) => (
   318→            <UserCard
   319→              key={u.id}
   320→              u={u}
   321→              locationMap={locationMap}
   322→              toggling={togglingId === u.id}
   323→              onToggle={() => toggleStatus(u)}
   324→              onEdit={() => openEdit(u)}
   325→            />
   326→          ))
   327→        )}
   328→      </div>
   329→
   330→      {/* Role legend */}
   331→      <RoleLegend />
   332→
   333→      {/* Create / Edit dialog */}
   334→      <UserDialog
   335→        open={dialogOpen}
   336→        onOpenChange={setDialogOpen}
   337→        editing={editing}
   338→      />
   339→    </div>
   340→  );
   341→}
   342→
   343→function UserTableRow({
   344→  u, locationMap, toggling, onToggle, onEdit,
   345→}: {
   346→  u: UserRow;
   347→  locationMap: Map<string, string>;
   348→  toggling: boolean;
   349→  onToggle: () => void;
   350→  onEdit: () => void;
   351→}) {
   352→  const assignedCities = u.assignedLocationIds
   353→    .map((id) => locationMap.get(id))
   354→    .filter(Boolean) as string[];
   355→
   356→  return (
   357→    <TableRow className="hover:bg-accent/30">
   358→      <TableCell className="pl-5">
   359→        <div className="flex items-center gap-2.5 min-w-0">
   360→          <Avatar className="size-8">
   361→            {u.avatar ? <AvatarImage src={u.avatar} alt={u.name} /> : null}
   362→            <AvatarFallback className={cn("text-[10px] font-semibold", ROLE_BADGE[u.role].split(" ").slice(0, 2).join(" "))}>
   363→              {initials(u.name)}
   364→            </AvatarFallback>
   365→          </Avatar>
   366→          <span className="font-medium truncate">{u.name}</span>
   367→        </div>
   368→      </TableCell>
   369→      <TableCell className="text-muted-foreground">{u.email}</TableCell>
   370→      <TableCell>
   371→        <RoleBadge role={u.role} />
   372→      </TableCell>
   373→      <TableCell>
   374→        {u.role === "branch_manager" ? (
   375→          assignedCities.length === 0 ? (
   376→            <span className="text-xs text-muted-foreground italic">Unassigned</span>
   377→          ) : (
   378→            <div className="flex flex-wrap gap-1 max-w-[220px]">
   379→              {assignedCities.slice(0, 3).map((c) => (
   380→                <Badge key={c} variant="outline" className="text-[10px] px-1.5 py-0 font-normal">
   381→                  {c}
   382→                </Badge>
   383→              ))}
   384→              {assignedCities.length > 3 && (
   385→                <Badge variant="outline" className="text-[10px] px-1.5 py-0 font-normal">
   386→                  +{assignedCities.length - 3}
   387→                </Badge>
   388→              )}
   389→            </div>
   390→          )
   391→        ) : (
   392→          <span className="text-xs text-muted-foreground">All</span>
   393→        )}
   394→      </TableCell>
   395→      <TableCell>
   396→        <div className="flex items-center gap-2">
   397→          <Switch checked={u.status === "active"} onCheckedChange={onToggle} disabled={toggling} aria-label="Toggle active" />
   398→          {u.status === "active" ? (
   399→            <span className="text-xs text-emerald-600 dark:text-emerald-400 font-medium">Active</span>
   400→          ) : (
   401→            <span className="text-xs text-muted-foreground">Inactive</span>
   402→          )}
   403→        </div>
   404→      </TableCell>
   405→      <TableCell className="text-xs text-muted-foreground tabular-nums">
   406→        {u.lastLoginAt ? formatDistanceToNow(new Date(u.lastLoginAt), { addSuffix: true }) : "Never"}
   407→      </TableCell>
   408→      <TableCell className="text-xs text-muted-foreground tabular-nums">
   409→        {format(new Date(u.createdAt), "dd MMM yyyy")}
   410→      </TableCell>
   411→      <TableCell className="pr-5 text-right">
   412→        <Button variant="ghost" size="sm" onClick={onEdit} className="h-7 px-2">
   413→          <Pencil className="size-3.5 mr-1" /> Edit
   414→        </Button>
   415→      </TableCell>
   416→    </TableRow>
   417→  );
   418→}

   581→function UserDialog({
   582→  open, onOpenChange, editing,
   583→}: {
   584→  open: boolean;
   585→  onOpenChange: (o: boolean) => void;
   586→  editing: UserRow | null;
   587→}) {
   588→  const qc = useQueryClient();
   589→  const { data: locations, isLoading: locationsLoading } = useLocations();
   590→
   591→  const isEdit = !!editing;
   592→  const [name, setName] = useState("");
   593→  const [email, setEmail] = useState("");
   594→  const [role, setRole] = useState<Role>("viewer");
   595→  const [password, setPassword] = useState("");
   596→  const [assignedIds, setAssignedIds] = useState<string[]>([]);
   597→  const [active, setActive] = useState(true);
   598→  const [errors, setErrors] = useState<Record<string, string>>({});
   599→  const [saving, setSaving] = useState(false);
   600→
   601→  // Sync form when dialog opens
   602→  useEffect(() => {
   603→    if (!open) return;
   604→    if (editing) {
   605→      setName(editing.name);
   606→      setEmail(editing.email);
   607→      setRole(editing.role);
   608→      setPassword("");
   609→      setAssignedIds(editing.assignedLocationIds ?? []);
   610→      setActive(editing.status === "active");
   611→    } else {
   612→      setName("");
   613→      setEmail("");
   614→      setRole("viewer");
   615→      setPassword("");
   616→      setAssignedIds([]);
   617→      setActive(true);
   618→    }
   619→    setErrors({});
   620→  }, [open, editing]);
   621→
   622→  function toggleLocation(id: string) {
   623→    setAssignedIds((prev) =>
   624→      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id],
   625→    );

   628→  function validate(): boolean {
   629→    const e: Record<string, string> = {};

   631→    if (!email.trim()) e.email = "Email is required";
   632→    else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) e.email = "Invalid email";
   633→    if (!isEdit && !password) e.password = "Password is required";
   634→    else if (!isEdit && password.length < 8) e.password = "Min 8 characters";
   635→    if (role === "branch_manager" && assignedIds.length === 0) {
   636→      e.assigned = "Assign at least one location";
   637→    }
   638→    setErrors(e);
   639→    return Object.keys(e).length === 0;
   640→  }

   643→    if (!validate()) return;
   644→    setSaving(true);

   646→      if (isEdit) {
   647→        await api("/api/users", {
   648→          method: "PATCH",
   649→          body: JSON.stringify({
   650→            id: editing!.id,
   651→            name: name.trim(),
   652→            role,
   653→            active,
   654→            assignedLocationIds: role === "branch_manager" ? assignedIds : [],
   655→          }),
   656→        });
   657→        toast.success("User updated");

   659→        await api("/api/users", {
   660→          method: "POST",
   661→          body: JSON.stringify({
   662→            name: name.trim(),
   663→            email: email.trim(),
   664→            role,
   665→            password,
   666→            assignedLocationIds: role === "branch_manager" ? assignedIds : [],
   667→          }),
   668→        });
   669→        toast.success("User invited");
   670→      }
   671→      qc.invalidateQueries({ queryKey: ["users"] });
   672→      onOpenChange(false);
   673→    } catch (e: any) {
   674→      toast.error(e?.message || "Failed to save user");
   675→    } finally {
   676→      setSaving(false);
   677→    }
   678→  }

   680→  return (
   681→    <Dialog open={open} onOpenChange={onOpenChange}>
   682→      <DialogContent className="sm:max-w-md max-h-[90vh] overflow-y-auto scroll-area">
   683→        <DialogHeader>
   684→          <DialogTitle>{isEdit ? "Edit user" : "Invite a new user"}</DialogTitle>
   685→          <DialogDescription>
   686→            {isEdit
   687→              ? "Update role, location assignments, and status."
   688→              : "Create a new team member and assign a role."}
   689→          </DialogDescription>
   690→        </DialogHeader>
   691→
   692→        <div className="space-y-4">
   693→          {/* Name */}
   694→          <div className="space-y-1.5">
   695→            <Label htmlFor="u-name">Full name</Label>
   696→            <Input
   697→              id="u-name"
   698→              value={name}
   699→              onChange={(e) => setName(e.target.value)}
   700→              placeholder="e.g. Priya Sharma"
   701→              aria-invalid={!!errors.name}
   702→            />
   703→            {errors.name && <p className="text-xs text-destructive">{errors.name}</p>}
   704→          </div>
   705→
   706→          {/* Email */}
   707→          <div className="space-y-1.5">
   708→            <Label htmlFor="u-email">Email</Label>
   709→            <Input
   710→              id="u-email"
   711→              type="email"
   712→              value={email}
   713→              onChange={(e) => setEmail(e.target.value)}
   714→              placeholder="name@myfng.in"
   715→              aria-invalid={!!errors.email}
   716→              disabled={isEdit}
   717→            />
   718→            {errors.email && <p className="text-xs text-destructive">{errors.email}</p>}
   719→            {isEdit && (
   720→              <p className="text-[11px] text-muted-foreground">Email cannot be changed after creation.</p>
   721→            )}
   722→          </div>
   723→
   724→          {/* Password (create only) */}
   725→          {!isEdit && (
   726→            <div className="space-y-1.5">
   727→              <Label htmlFor="u-pass">Password</Label>
   728→              <Input
   729→                id="u-pass"
   730→                type="password"
   731→                value={password}
   732→                onChange={(e) => setPassword(e.target.value)}
   733→                placeholder="Min 8 characters"
   734→                aria-invalid={!!errors.password}
   735→              />
   736→              {errors.password && <p className="text-xs text-destructive">{errors.password}</p>}
   737→            </div>
   738→          )}
   739→
   740→          {/* Role */}
   741→          <div className="space-y-1.5">
   742→            <Label>Role</Label>
   743→            <Select value={role} onValueChange={(v) => setRole(v as Role)}>
   744→              <SelectTrigger className="w-full">
   745→                <SelectValue />
   746→              </SelectTrigger>
   747→              <SelectContent>
   748→                {ROLES.map((r) => (
   749→                  <SelectItem key={r.value} value={r.value}>
   750→                    {r.label}
   751→                  </SelectItem>
   752→                ))}
   753→              </SelectContent>
   754→            </Select>
   755→            <p className="text-[11px] text-muted-foreground leading-snug">
   756→              {ROLES.find((r) => r.value === role)?.description}
   757→            </p>
   758→          </div>
   759→
   760→          {/* Assigned locations — only for branch_manager */}
   761→          {role === "branch_manager" && (
   762→            <div className="space-y-1.5">
   763→              <Label>Assigned locations</Label>
   764→              <div className="rounded-md border max-h-44 overflow-y-auto scroll-area divide-y">
   765→                {locationsLoading ? (
   766→                  <div className="p-3 space-y-2">
   767→                    {Array.from({ length: 3 }).map((_, i) => (
   768→                      <Skeleton key={i} className="h-5 w-full" />
   769→                    ))}
   770→                  </div>
   771→                ) : (locations ?? []).length === 0 ? (
   772→                  <p className="text-xs text-muted-foreground p-3">No locations available.</p>
   773→                ) : (
   774→                  (locations ?? []).map((l) => (
   775→                    <label
   776→                      key={l.id}
   777→                      className="flex items-center gap-2.5 px-3 py-2 hover:bg-accent/40 cursor-pointer"
   778→                    >
   779→                      <Checkbox
   780→                        checked={assignedIds.includes(l.id)}
   781→                        onCheckedChange={() => toggleLocation(l.id)}

   783→                      <span className="text-sm">{l.name}</span>

   806→        </div>
   807→
   808→        <DialogFooter>
   809→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>
   810→            Cancel
   811→          </Button>

   813→            {saving ? (

   816→              <>{isEdit ? "Save changes" : "Invite user"}</>
   817→            )}
   818→          </Button>
   819→        </DialogFooter>
   820→      </DialogContent>
   821→    </Dialog>
   822→  );
   823→}
