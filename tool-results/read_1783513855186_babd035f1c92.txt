   326→export function SettingsView() {

   330→
   331→  // Resolve the active category — if unset or hidden, fall back to the first visible one.
   332→  const validActive: CategoryKey = active && visible.find((c) => c.key === active)
   333→    ? active
   334→    : (visible[0]?.key ?? "overview");
   335→
   336→  return (
   337→    <div className="p-4 sm:p-6 lg:p-8 space-y-5">
   338→      <PageHeader
   339→        title="Settings"
   340→        description="Manage users, integrations, security & system configuration"
   341→        icon={Settings}
   342→      />
   343→
   344→      <div className="flex flex-col md:flex-row gap-4 md:gap-6">
   345→        {/* Desktop sidebar */}
   346→        <aside className="hidden md:block w-56 shrink-0">
   347→          <nav className="space-y-1 sticky top-4">
   348→            {visible.map((cat) => {
   349→              const isActive = validActive === cat.key;
   350→              const Icon = cat.icon;
   351→              return (
   352→                <button
   353→                  key={cat.key}
   354→                  onClick={() => setActive(cat.key)}
   355→                  className={cn(
   356→                    "w-full flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm font-medium transition-colors text-left",
   357→                    isActive
   358→                      ? "bg-primary text-primary-foreground shadow-sm"
   359→                      : "text-muted-foreground hover:bg-accent hover:text-foreground",
   360→                  )}
   361→                >
   362→                  <Icon className="size-4 shrink-0" />
   363→                  <span className="truncate">{cat.label}</span>
   364→                </button>
   365→              );
   366→            })}
   367→          </nav>
   368→        </aside>
   369→
   370→        {/* Mobile category picker — horizontal scrollable pills */}
   371→        <div className="md:hidden w-full">
   372→          <div className="flex gap-2 overflow-x-auto scroll-area pb-2 -mx-1 px-1">
   373→            {visible.map((cat) => {
   374→              const isActive = validActive === cat.key;
   375→              const Icon = cat.icon;
   376→              return (
   377→                <button
   378→                  key={cat.key}
   379→                  onClick={() => setActive(cat.key)}
   380→                  className={cn(
   381→                    "flex items-center gap-1.5 px-3 py-2 rounded-lg text-xs font-medium transition-colors shrink-0 whitespace-nowrap",
   382→                    isActive
   383→                      ? "bg-primary text-primary-foreground shadow-sm"

   387→                  <Icon className="size-3.5 shrink-0" />

   429→function OverviewContent() {
   430→  const user = useUser();
   431→  const { data: users } = useQuery<UserRow[]>({
   432→    queryKey: ["users"],
   433→    queryFn: () => api<UserRow[]>("/api/users"),
   434→  });
   435→  const { data: locations } = useLocations();
   436→  const { data: health, isLoading: healthLoading } = useQuery<HealthResponse>({
   437→    queryKey: ["admin", "system-health"],
   438→    queryFn: () => api<HealthResponse>("/api/admin/system-health"),
   439→  });
   440→  const { data: jobs } = useQuery<JobsResponse>({
   441→    queryKey: ["admin", "jobs"],
   442→    queryFn: () => api<JobsResponse>("/api/admin/jobs"),
   443→  });
   444→  const canSystem = can(user.role, "system.view");
   445→  const { data: system } = useQuery<SystemResponse>({
   446→    queryKey: ["system"],
   447→    queryFn: () => api<SystemResponse>("/api/system"),
   448→    enabled: canSystem,
   449→  });
   450→
   451→  const activeLocations = (locations ?? []).filter((l) => l.status === "active").length;
   452→  const totalUsers = users?.length ?? 0;
   453→  const failedJobs = jobs?.stats.failed ?? 0;
   454→  const pendingAiJobs = jobs?.jobs.filter((j) => j.queueName === "ai-processing" && (j.status === "queued" || j.status === "processing")).length ?? 0;
   455→  const dbCheck = health?.checks.find((c) => c.service === "Database");
   456→  const storageUsed = system?.storageBuckets.reduce((a, b) => a + b.totalSize, 0) ?? 0;
   457→  const storageUsedMb = (storageUsed / 1024 / 1024).toFixed(1);
   458→
   459→  const errorLogs = system?.errorLogs ?? [];
   460→
   461→  return (
   462→    <div className="space-y-4">
   463→      <div>
   464→        <h2 className="text-lg font-semibold">Admin Dashboard</h2>
   465→        <p className="text-sm text-muted-foreground mt-0.5">Real-time system health, recent alerts & operational overview.</p>
   466→      </div>
   467→
   468→      {/* Stat cards */}
   469→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   470→        <StatCard label="Total Users" value={totalUsers} icon={Users} accent="emerald" hint="Across all roles" />
   471→        <StatCard label="Active Locations" value={activeLocations} icon={Building2} accent="teal" hint={`of ${locations?.length ?? 0} total`} />
   472→        <StatCard label="Google Accounts" value="1" icon={Plug} accent="amber" hint="OAuth connected" />
   473→        <StatCard
   474→          label="System Health"
   475→          value={health ? health.overall.charAt(0).toUpperCase() + health.overall.slice(1) : "—"}
   476→          icon={HeartPulse}
   477→          accent={health?.overall === "healthy" ? "emerald" : health?.overall === "warning" ? "amber" : "rose"}
   478→          hint={health ? `${health.summary.healthy}/${health.summary.total} services healthy` : "Loading…"}
   479→        />
   480→        <StatCard label="Failed Jobs" value={failedJobs} icon={AlertTriangle} accent={failedJobs > 0 ? "rose" : "emerald"} hint="Requires attention" />
   481→        <StatCard label="Pending AI Jobs" value={pendingAiJobs} icon={Sparkles} accent="amber" hint="Queued + processing" />
   482→        <StatCard label="Storage Used" value={`${storageUsedMb} MB`} icon={HardDrive} accent="teal" hint="Across all buckets" />
   483→        <StatCard
   484→          label="Database"
   485→          value={dbCheck ? dbCheck.status === "healthy" ? "Healthy" : dbCheck.status === "warning" ? "Slow" : "Down" : "—"}
   486→          icon={Database}
   487→          accent={dbCheck?.status === "healthy" ? "emerald" : dbCheck?.status === "warning" ? "amber" : "rose"}
   488→          hint={dbCheck?.latency ? `${dbCheck.latency}ms latency` : "—"}
   489→        />
   490→      </div>
   491→
   492→      {/* System health summary */}
   493→      <CardSection
   494→        title="System Health Summary"
   495→        description="Aggregate status across all backend services"
   496→        action={
   497→          health ? (
   498→            <Badge variant="outline" className={cn("text-xs font-medium", healthColor(health.overall))}>
   499→              <span className={cn("size-1.5 rounded-full mr-1", healthDot(health.overall))} />
   500→              {health.overall.toUpperCase()}
   501→            </Badge>
   502→          ) : null
   503→        }
   504→      >
   505→        {healthLoading ? (

   997→function UserDialog({
   998→  open, onOpenChange, editing,
   999→}: {
  1000→  open: boolean;
  1001→  onOpenChange: (o: boolean) => void;
  1002→  editing: UserRow | null;
  1003→}) {
  1004→  const qc = useQueryClient();
  1005→  const { data: locations, isLoading: locationsLoading } = useLocations();
  1006→
  1007→  const isEdit = !!editing;
  1008→  const [name, setName] = useState("");
  1009→  const [email, setEmail] = useState("");
  1010→  const [role, setRole] = useState<Role>("viewer");
  1011→  const [password, setPassword] = useState("");
  1012→  const [assignedIds, setAssignedIds] = useState<string[]>([]);
  1013→  const [active, setActive] = useState(true);
  1014→  const [errors, setErrors] = useState<Record<string, string>>({});
  1015→  const [saving, setSaving] = useState(false);
  1016→
  1017→  useEffect(() => {
  1018→    if (!open) return;
  1019→    if (editing) {
  1020→      setName(editing.name);
  1021→      setEmail(editing.email);
  1022→      setRole(editing.role);
  1023→      setPassword("");
  1024→      setAssignedIds(editing.assignedLocationIds ?? []);
  1025→      setActive(editing.status === "active");
  1026→    } else {
  1027→      setName("");
  1028→      setEmail("");
  1029→      setRole("viewer");
  1030→      setPassword("");
  1031→      setAssignedIds([]);
  1032→      setActive(true);
  1033→    }
  1034→    setErrors({});
