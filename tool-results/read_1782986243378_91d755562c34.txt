    68→function initials(name: string) {
    69→  return (
    70→    name
    71→      .split(" ")
    72→      .map((p) => p[0])
    73→      .filter(Boolean)
    74→      .slice(0, 2)
    75→      .join("")
    76→      .toUpperCase() || "?"
    77→  );
    78→}
    79→
    80→function colorFor(name: string) {
    81→  let h = 0;
    82→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
    83→  return AVATAR_COLORS[h % AVATAR_COLORS.length];
    84→}

   502→                MiSA AI drafts are suggestions — review before publishing.

   542→            </Button>
   543→          </DialogFooter>
   544→        </DialogContent>
   545→      </Dialog>
   546→    </div>
   547→  );
   548→}

   552→function ReviewCard({
   553→  review,
   554→  canReply,
   555→  canAiReply,
   556→  aiLoading,
   557→  onAiDraft,
   558→  onReply,
   559→  onIgnore,
   560→  ignoring,
   561→}: {
   562→  review: ReviewWithLocation;
   563→  canReply: boolean;
   564→  canAiReply: boolean;
   565→  aiLoading: boolean;
   566→  onReply: () => void;
   567→  onAiDraft: () => void;
   568→  onIgnore: () => void;
   569→  ignoring: boolean;
   570→}) {
   571→  const [expanded, setExpanded] = useState(false);
   572→  const isNegative = review.rating <= 2;
   573→  const hasReply =
   574→    !!review.replyText && review.replyStatus === "replied";
   575→  const isLong = review.text.length > 180;
   576→
   577→  return (
   578→    <Card
   579→      className={cn(
   580→        "p-0 overflow-hidden transition-shadow hover:shadow-md",
   581→        isNegative && "border-l-4 border-l-rose-500",
   582→      )}
   583→    >
   584→      <CardContent className="p-4 space-y-3">
   585→        {/* header */}
   586→        <div className="flex items-start gap-3">
   587→          <Avatar className="size-10">
   588→            {review.authorPhoto && (
   589→              <AvatarImage src={review.authorPhoto} alt={review.authorName} />
   590→            )}
   591→            <AvatarFallback className={colorFor(review.authorName)}>
   592→              {initials(review.authorName)}
   593→            </AvatarFallback>
   594→          </Avatar>
   595→          <div className="min-w-0 flex-1">
   596→            <div className="flex items-start justify-between gap-2">
   597→              <div className="min-w-0">
   598→                <div className="text-sm font-semibold truncate">
   599→                  {review.authorName}
   600→                </div>
   601→                <div className="text-xs text-muted-foreground truncate">
   602→                  {review.locationName} · {review.locationCity}
   603→                </div>
   604→              </div>
   605→              <RatingStars rating={review.rating} size={16} />
   606→            </div>
   607→            <div className="text-[11px] text-muted-foreground mt-0.5">
   608→              {formatDistanceToNow(new Date(review.createdAt), {
   609→                addSuffix: true,
   610→              })}
   611→            </div>
   612→          </div>
   613→        </div>
   614→
   615→        {/* review text */}
   616→        <div>
   617→          <p
   618→            className={cn(
   619→              "text-sm leading-relaxed text-foreground/90 whitespace-pre-line",
   620→              !expanded && "line-clamp-4",
   621→            )}
   622→          >
   623→            {review.text}
   624→          </p>

   626→            <button
   627→              type="button"

   629→              className="text-xs text-primary hover:underline mt-1"
   630→            >
   631→              {expanded ? "Show less" : "Show more"}
   632→            </button>
   633→          )}
   634→        </div>
   635→
   636→        {/* tags */}
   637→        <div className="flex flex-wrap items-center gap-2">
   638→          <SentimentBadge sentiment={review.sentiment} />
   639→          <ReplyStatusBadge
   640→            status={review.replyStatus}
   641→            source={review.replySource}
   642→          />
   643→        </div>
   644→
   645→        {/* existing reply */}
   646→        {hasReply && review.replyText && (
   647→          <div className="rounded-lg bg-muted/40 border border-muted p-3">
   648→            <div className="flex items-center gap-1.5 mb-1 flex-wrap">
   649→              <MessageSquare className="size-3.5 text-muted-foreground" />
   650→              <span className="text-[11px] font-medium text-muted-foreground">
   651→                {review.replySource === "ai"
   652→                  ? "Replied by MiSA AI"
   653→                  : "Replied manually"}
   654→              </span>
   655→              {review.replySource === "ai" && (
   656→                <Badge
   657→                  variant="outline"
   658→                  className="text-[10px] bg-amber-500/10 text-amber-700 dark:text-amber-400 border-amber-500/20"
   659→                >
   660→                  <Sparkles className="size-2.5 mr-0.5" /> AI
   661→                </Badge>
   662→              )}
   663→              {review.repliedAt && (
   664→                <span className="text-[10px] text-muted-foreground ml-auto">
   665→                  {formatDistanceToNow(new Date(review.repliedAt), {
   666→                    addSuffix: true,
   667→                  })}
   668→                </span>
   669→              )}
   670→            </div>
   671→            <p className="text-xs text-foreground/80 leading-relaxed line-clamp-3">
   672→              {review.replyText}
   673→            </p>
   674→          </div>
   675→        )}
   676→
   677→        {/* actions */}
   678→        {canReply && (
   679→          <div className="flex flex-wrap items-center gap-2 pt-1 border-t">
   680→            {canAiReply && (
   681→              <Button
   682→                size="sm"
   683→                variant="outline"
   684→                onClick={onAiDraft}
   685→                disabled={aiLoading}
   686→                className="min-h-11 border-amber-500/40 text-amber-700 dark:text-amber-400 hover:bg-amber-500/10"
   687→              >
   688→                {aiLoading ? (
   689→                  <>
   690→                    <Loader2 className="size-3.5 mr-1.5 animate-spin" /> MiSA
   691→                    AI…
   692→                  </>
   693→                ) : (
   694→                  <>
   695→                    <Sparkles className="size-3.5 mr-1.5" /> MiSA AI draft
   696→                  </>
   697→                )}
   698→              </Button>
   699→            )}
   700→            <Button
   701→              size="sm"
   702→              variant="outline"
   703→              onClick={onReply}
   704→              className="min-h-11"
   705→            >
   706→              <MessageSquare className="size-3.5 mr-1.5" />{" "}
   707→              {hasReply ? "Edit reply" : "Reply"}
   708→            </Button>
   709→            {review.replyStatus !== "ignored" && (
   710→              <Button
   711→                size="sm"
   712→                variant="ghost"
   713→                onClick={onIgnore}
   714→                disabled={ignoring}
   715→                className="min-h-11 ml-auto text-muted-foreground hover:text-rose-600"
   716→              >
   717→                {ignoring ? (
   718→                  <Loader2 className="size-3.5 mr-1.5 animate-spin" />
   719→                ) : (
   720→                  <EyeOff className="size-3.5 mr-1.5" />
   721→                )}
   722→                Ignore
   723→              </Button>
   724→            )}
   725→          </div>
   726→        )}
   727→      </CardContent>
   728→    </Card>
   729→  );
   730→}
