    94→    94→function relativeTime(iso: string | null): string | null {
    95→    95→  if (!iso) return null;
    96→    96→  try {
    97→    97→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
    98→    98→  } catch {
    99→    99→    return null;
   100→   100→  }
   101→   101→}
   102→   102→
   103→   103→function scheduleLabel(iso: string | null): string {
   104→   104→  if (!iso) return "";
   105→   105→  try {
   106→   106→    const d = new Date(iso);
   107→   107→    if (isToday(d)) return `Today, ${format(d, "h:mm a")}`;
   108→   108→    if (isTomorrow(d)) return `Tomorrow, ${format(d, "h:mm a")}`;
   109→   109→    return format(d, "d MMM yyyy, h:mm a");
   110→   110→  } catch {
   111→   111→    return "";
   112→   112→  }
   113→   113→}

   355→   355→
   356→   356→function PostCard({
   357→   357→  post, canManage, onPublish, onEdit, onDelete,
   358→   358→}: {
   359→   359→  post: PostWithLocation;
   360→   360→  canManage: boolean;
   361→   361→  onPublish: (p: PostWithLocation) => void;
   362→   362→  onEdit: (p: PostWithLocation) => void;
   363→   363→  onDelete: (p: PostWithLocation) => void;
   364→   364→}) {
   365→   365→  const meta = postTypeMeta(post.type);
   366→   366→  const Icon = meta.icon;
   367→   367→  const [scheduleOpen, setScheduleOpen] = React.useState(false);
   368→   368→  const [scheduleDate, setScheduleDate] = React.useState<Date | null>(
   369→   369→    post.scheduledAt ? new Date(post.scheduledAt) : null,
   370→   370→  );
   371→   371→  const [scheduling, setScheduling] = React.useState(false);
   372→   372→  const qc = useQueryClient();
   373→   373→
   374→   374→  const relPublished = relativeTime(post.publishedAt);
   375→   375→  const relScheduled = relativeTime(post.scheduledAt);
   376→   376→  const relCreated = relativeTime(post.createdAt);
   377→   377→
   378→   378→  async function applySchedule() {
   379→   379→    if (!scheduleDate) return;
   380→   380→    setScheduling(true);
   381→   381→    try {
   382→   382→      await api(`/api/posts/${post.id}`, {
   383→   383→        method: "PATCH",
   384→   384→        body: JSON.stringify({ status: "scheduled", scheduledAt: scheduleDate.toISOString() }),
   385→   385→      });
   386→   386→      qc.invalidateQueries({ queryKey: ["posts"] });
   387→   387→      toast.success(`Scheduled for ${scheduleLabel(scheduleDate.toISOString())}`);
   388→   388→      setScheduleOpen(false);
   389→   389→    } catch (e: any) {
   390→   390→      toast.error(e?.message || "Failed to schedule");
   391→   391→    } finally {
   392→   392→      setScheduling(false);
   393→   393→    }
   394→   394→  }
   395→   395→
   396→   396→  return (
   397→   397→    <Card className="group relative hover:shadow-md transition-shadow overflow-hidden">
   398→   398→      <CardContent className="p-4 space-y-3">
   399→   399→        {/* Header: type + actions */}
   400→   400→        <div className="flex items-start justify-between gap-2">
   401→   401→          <div className="flex items-center gap-2 min-w-0">
   402→   402→            <span className={cn("size-8 rounded-lg flex items-center justify-center shrink-0", meta.tint)}>
   403→   403→              <Icon className="size-4" />
   404→   404→            </span>
   405→   405→            <div className="min-w-0">
   406→   406→              <div className="text-[11px] font-medium text-muted-foreground uppercase tracking-wide">
   407→   407→                {meta.label}
   408→   408→              </div>
   409→   409→              <div className="text-[10px] text-muted-foreground truncate flex items-center gap-1">
   410→   410→                <MapPin className="size-2.5" />
   411→   411→                {post.locationName}
   412→   412→              </div>
   413→   413→            </div>
   414→   414→          </div>
   415→   415→
   416→   416→          {canManage && (
   417→   417→            <DropdownMenu>
   418→   418→              <DropdownMenuTrigger asChild>
   419→   419→                <Button variant="ghost" size="icon" className="size-7 -mt-1 -mr-1 opacity-60 group-hover:opacity-100 transition-opacity">
   420→   420→                  <MoreVertical className="size-4" />
   421→   421→                </Button>
   422→   422→              </DropdownMenuTrigger>
   423→   423→              <DropdownMenuContent align="end" className="w-44">
   424→   424→                {(post.status === "draft" || post.status === "scheduled" || post.status === "failed") && (
   425→   425→                  <DropdownMenuItem onClick={() => onPublish(post)}>
   426→   426→                    <Send className="size-3.5 mr-2 text-emerald-500" /> Publish now
   427→   427→                  </DropdownMenuItem>
   428→   428→                )}
   429→   429→                <DropdownMenuItem onClick={() => setScheduleOpen(true)}>
   430→   430→                  <CalendarClock className="size-3.5 mr-2 text-amber-500" /> Schedule…
   431→   431→                </DropdownMenuItem>
   432→   432→                <DropdownMenuItem onClick={() => onEdit(post)}>
   433→   433→                  <Pencil className="size-3.5 mr-2" /> Edit
   434→   434→                </DropdownMenuItem>
   435→   435→                <DropdownMenuSeparator />
   436→   436→                <DropdownMenuItem
   437→   437→                  className="text-rose-600 dark:text-rose-400 focus:text-rose-700"
   438→   438→                  onClick={() => onDelete(post)}
