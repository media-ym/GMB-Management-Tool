    24→    24→export function getGoogleAuthUrl(state?: string): { url: string; state: string } {
    25→    25→  // Generate a high-entropy CSRF state if none provided.
    26→    26→  // The caller is responsible for storing this value (typically in a cookie)
    27→    27→  // and validating it on the OAuth callback.
    28→    28→  const finalState = state || crypto.randomUUID();
    29→    29→  const params = new URLSearchParams({
    30→    30→    client_id: GOOGLE_CLIENT_ID,
    31→    31→    redirect_uri: GOOGLE_REDIRECT_URI,
    32→    32→    response_type: "code",
    33→    33→    scope: GBP_SCOPES,
    34→    34→    access_type: "offline",
    35→    35→    prompt: "consent",
    36→    36→    state: finalState,
    37→    37→  });
    38→    38→  return { url: `https://accounts.google.com/o/oauth2/v2/auth?${params.toString()}`, state: finalState };
    39→    39→}

   100→   100→export async function getValidAccessToken(): Promise<string | null> {
   101→   101→  const account = await db.googleAccount.findFirst();
   102→   102→  if (!account) return null;
   103→   103→
   104→   104→  // Defensive: if the stored token is the legacy "no_token" placeholder, treat as no token.
   105→   105→  if (account.accessToken === "no_token") return null;
   106→   106→
   107→   107→  // Check if token is still valid (5 min buffer)
   108→   108→  if (account.tokenExpiry && new Date(account.tokenExpiry) > new Date(Date.now() + 5 * 60 * 1000)) {
   109→   109→    const decrypted = decryptToken(account.accessToken);
   110→   110→    if (!decrypted) {
   111→   111→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   112→   112→      return null;
   113→   113→    }
   114→   114→    return decrypted;
   115→   115→  }
   116→   116→
   117→   117→  // Try to refresh — decrypt the refresh token first
   118→   118→  if (account.refreshToken) {
   119→   119→    const decryptedRefresh = decryptToken(account.refreshToken);
   120→   120→    if (!decryptedRefresh) {
   121→   121→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   122→   122→      return null;
   123→   123→    }
   124→   124→    try {
   125→   125→      const { accessToken, expiryDate } = await refreshAccessToken(decryptedRefresh);
   126→   126→      await db.googleAccount.update({
   127→   127→        where: { id: account.id },
   128→   128→        data: {
   129→   129→          accessToken: encryptToken(accessToken), // ENCRYPT before saving
   130→   130→          tokenExpiry: new Date(expiryDate),
   131→   131→          status: "active",
   132→   132→        },
   133→   133→      });
   134→   134→      return accessToken; // return decrypted
   135→   135→    } catch (e) {
   136→   136→      console.error("Token refresh failed:", e);
   137→   137→      await db.googleAccount.update({ where: { id: account.id }, data: { status: "expired" } });
   138→   138→      return null;
   139→   139→    }
   140→   140→  }
   141→   141→
   142→   142→  return null;
   143→   143→}

   321→   321→// Fetch ALL 5 metric types for a date range and return as daily aggregates
   322→   322→export async function getFullPerformanceMetrics(
   323→   323→  accessToken: string,
   324→   324→  locationName: string,
   325→   325→  daysBack: number = 30
   326→   326→): Promise<{ date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }[]> {
   327→   327→  const today = new Date();
   328→   328→  const start = new Date(today);
   329→   329→  start.setDate(today.getDate() - daysBack);
   330→   330→
   331→   331→  const startDate = { year: start.getFullYear(), month: start.getMonth() + 1, day: start.getDate() };
   332→   332→  const endDate = { year: today.getFullYear(), month: today.getMonth() + 1, day: today.getDate() };
   333→   333→
   334→   334→  // Google API returns one metric type per call, so fetch all 5 in parallel
   335→   335→  const metricTypes = {
   336→   336→    searchViews: "BUSINESS_IMPRESSIONS_DESKTOP_SEARCH",
   337→   337→    mapsViews: "BUSINESS_IMPRESSIONS_DESKTOP_MAPS",
   338→   338→    websiteClicks: "WEBSITE_CLICKS",
   339→   339→    phoneCalls: "CALL_CLICKS",
   340→   340→    directionRequests: "BUSINESS_DIRECTION_REQUESTS",
   341→   341→  };
   342→   342→
   343→   343→  const [searchRes, mapsRes, websiteRes, callRes, directionRes] = await Promise.allSettled([
   344→   344→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.searchViews),
   345→   345→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.mapsViews),
   346→   346→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.websiteClicks),
   347→   347→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.phoneCalls),
   348→   348→    getPerformanceMetrics(accessToken, locationName, startDate, endDate, metricTypes.directionRequests),
   349→   349→  ]);
   350→   350→
   351→   351→  // Build a map of date → metrics
   352→   352→  const dailyMap = new Map<string, { date: Date; searchViews: number; mapsViews: number; websiteClicks: number; phoneCalls: number; directionRequests: number }>();
   353→   353→
   354→   354→  function populate(result: PromiseSettledResult<any[]>, key: keyof typeof metricTypes) {
   355→   355→    if (result.status !== "fulfilled") return;
   356→   356→    for (const entry of result.value) {
   357→   357→      const dateStr = entry.date?.year
   358→   358→        ? `${entry.date.year}-${String(entry.date.month).padStart(2, "0")}-${String(entry.date.day).padStart(2, "0")}`
   359→   359→        : null;
   360→   360→      if (!dateStr) continue;
   361→   361→      if (!dailyMap.has(dateStr)) {
   362→   362→        dailyMap.set(dateStr, {
   363→   363→          date: new Date(dateStr),
   364→   364→          searchViews: 0, mapsViews: 0, websiteClicks: 0, phoneCalls: 0, directionRequests: 0,
   365→   365→        });
   366→   366→      }
   367→   367→      const day = dailyMap.get(dateStr)!;
   368→   368→      const value = entry.value ?? 0;
   369→   369→      (day as any)[key] += value;
   370→   370→    }
   371→   371→  }
   372→   372→
   373→   373→  populate(searchRes, "searchViews");
   374→   374→  populate(mapsRes, "mapsViews");
   375→   375→  populate(websiteRes, "websiteClicks");
   376→   376→  populate(callRes, "phoneCalls");
   377→   377→  populate(directionRes, "directionRequests");
   378→   378→
   379→   379→  return Array.from(dailyMap.values()).sort((a, b) => a.date.getTime() - b.date.getTime());
   380→   380→}

   391→   391→

   393→   393→    const dailyMetrics = await getFullPerformanceMetrics(accessToken, gbp.googleLocationId, daysBack);

   402→   402→        where: { locationId_date: { locationId, date: dateKey } },
   403→   403→        create: {
   404→   404→          locationId,
   405→   405→          date: dateKey,
   406→   406→          searchViews: day.searchViews,
   407→   407→          mapsViews: day.mapsViews,
   408→   408→          websiteClicks: day.websiteClicks,
   409→   409→          phoneCalls: day.phoneCalls,
   410→   410→          directionRequests: day.directionRequests,
   411→   411→          bookings: 0,
   412→   412→        },
   413→   413→        update: {

   432→   432→export async function updateGoogleBusinessProfile(
   433→   433→  accessToken: string,
   434→   434→  locationName: string,
   435→   435→  updates: {
   436→   436→    title?: string;
   437→   437→    phone?: string;
   438→   438→    website?: string;

   440→   440→    appointmentUrl?: string;

   442→   442→    categories?: { primaryCategory: string; additionalCategories?: string[] };

   453→   453→    fieldMask.push("profile.description");

   455→   455→  if (updates.appointmentUrl !== undefined) {

   463→   463→        openTime: h.openTime,

   465→   465→        closeTime: h.closeTime,

   472→   472→      primaryCategory: { displayName: updates.categories.primaryCategory },
   473→   473→      additionalCategories: (updates.categories.additionalCategories || []).map(c => ({ displayName: c })),
   474→   474→    };
   475→   475→    fieldMask.push("categories");
   476→   476→  }
   477→   477→
   478→   478→  const data = await withRetry<any>(() =>

   487→   487→  );
   488→   488→  return data;
   489→   489→}
