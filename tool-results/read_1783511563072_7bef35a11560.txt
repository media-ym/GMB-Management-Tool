   162→function relativeTime(iso: string | null): string {
   163→  if (!iso) return "Never";
   164→  try {
   165→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
   166→  } catch {
   167→    return "—";
   168→  }
   169→}
   170→
   171→function fullTime(iso: string | null): string {
   172→  if (!iso) return "—";
   173→  try {
   174→    return format(new Date(iso), "d MMM yyyy, h:mm a");
   175→  } catch {
   176→    return "—";
   177→  }
   178→}
   179→
   180→function scopeLabel(scope: string): string {
   181→  const found = REQUESTED_SCOPES.find((s) => s.scope === scope);
   182→  return found?.label ?? scope.replace(/^https:\/\/www\.googleapis\.com\/auth\/business\./, "").replace(/^https:\/\/www\.googleapis\.com\/auth\/userinfo\./, "userinfo.").replace(/^https:\/\/www\.googleapis\.com\/auth\//, "");
   183→}
   184→
   185→function tokenProgress(expiryIso: string | null): { pct: number; totalMs: number; remainingMs: number } {
   186→  if (!expiryIso) return { pct: 0, totalMs: 0, remainingMs: 0 };
   187→  const now = Date.now();
   188→  const expiry = new Date(expiryIso).getTime();
   189→  // assume access tokens are issued with a 1h lifetime
   190→  const totalMs = 60 * 60 * 1000;
   191→  const remainingMs = expiry - now;
   192→  const pct = Math.max(0, Math.min(100, (remainingMs / totalMs) * 100));
   193→  return { pct, totalMs, remainingMs };
   194→}

   260→function CopyButton({ value, label = "Copy" }: { value: string; label?: string }) {
   261→  const [copied, setCopied] = useState(false);
   262→  async function onCopy() {
   263→    try {
   264→      await navigator.clipboard.writeText(value);
   265→      setCopied(true);
   266→      toast.success("Copied to clipboard");
   267→      setTimeout(() => setCopied(false), 1500);
   268→    } catch {
   269→      toast.error("Couldn't copy to clipboard");
   270→    }
   271→  }
   272→  return (
   273→    <Button
   274→      type="button"
   275→      variant="ghost"
   276→      size="sm"
   277→      onClick={onCopy}
   278→      className="h-7 px-2 text-xs text-muted-foreground hover:text-foreground"
   279→    >
   280→      {copied ? <Check className="size-3.5 text-emerald-500" /> : <Copy className="size-3.5" />}
   281→      {copied ? "Copied" : label}
   282→    </Button>
   283→  );
   284→}
