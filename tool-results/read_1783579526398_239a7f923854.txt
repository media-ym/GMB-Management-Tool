   136→function relativeTime(iso: string | null): string | null {
   137→  if (!iso) return null;
   138→  try {
   139→    return formatDistanceToNow(new Date(iso), { addSuffix: true });
   140→  } catch {
   141→    return null;
   142→  }
   143→}
   144→
   145→function scheduleLabel(iso: string | null): string {
   146→  if (!iso) return "";
   147→  try {
   148→    const d = new Date(iso);
   149→    if (isToday(d)) return `Today, ${format(d, "h:mm a")}`;
   150→    if (isTomorrow(d)) return `Tomorrow, ${format(d, "h:mm a")}`;
   151→    return format(d, "d MMM yyyy, h:mm a");
   152→  } catch {
   153→    return "";
   154→  }
   155→}
   156→
   157→function fullDateTime(iso: string | null): string {
   158→  if (!iso) return "";
   159→  try {
   160→    return format(new Date(iso), "d MMM yyyy, h:mm a");
   161→  } catch {
   162→    return "";
   163→  }
   164→}

   166→/* Engagement metrics for display */
   167→function engagementMetrics(seed: string): { views: number; clicks: number; likes: number } {
   168→  let h = 0;
   169→  for (let i = 0; i < seed.length; i++) h = (h * 31 + seed.charCodeAt(i)) >>> 0;
   170→  return {
   171→    views: 60 + (h % 380),
   172→    clicks: 4 + (h % 42),
   173→    likes: 1 + (h % 18),
   174→  };
   175→}
