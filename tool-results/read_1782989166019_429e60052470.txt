    63→// ── Rank → color helpers ─────────────────────────────────────────────
    64→function rankColor(rank: number): { bg: string; text: string; label: string } {
    65→  if (!rank || rank <= 0) return { bg: "bg-slate-300 dark:bg-slate-700", text: "text-slate-600 dark:text-slate-300", label: "Not ranked" };
    66→  if (rank <= 3) return { bg: "bg-emerald-500", text: "text-white", label: "Top 3" };
    67→  if (rank <= 10) return { bg: "bg-amber-500", text: "text-white", label: "Top 10" };
    68→  if (rank <= 20) return { bg: "bg-orange-500", text: "text-white", label: "Top 20" };
    69→  return { bg: "bg-rose-500", text: "text-white", label: "21+" };
    70→}
    71→
    72→function rankColorSoft(rank: number): string {
    73→  if (!rank || rank <= 0) return "bg-slate-200 dark:bg-slate-700";
    74→  if (rank <= 3) return "bg-emerald-500";
    75→  if (rank <= 10) return "bg-amber-500";
    76→  if (rank <= 20) return "bg-orange-500";
    77→  return "bg-rose-500";
    78→}

    80→function rankTextClass(rank: number): string {
    81→  if (!rank || rank <= 0) return "text-slate-500";
    82→  if (rank <= 3) return "text-emerald-600 dark:text-emerald-400";
    83→  if (rank <= 10) return "text-amber-600 dark:text-amber-400";

    88→// Deterministic mock trend from a keyword string (seed data has no trend series)
    89→function mockTrend(keyword: string): { delta: number; dir: "up" | "down" | "flat" } {
    90→  let h = 0;
    91→  for (let i = 0; i < keyword.length; i++) h = (h * 31 + keyword.charCodeAt(i)) >>> 0;
    92→  const v = (h % 7) - 3; // -3..+3
    93→  return { delta: Math.abs(v), dir: v > 0 ? "up" : v < 0 ? "down" : "flat" };
    94→}

    96→// ── Build a normalized 5x5 display grid from a flat list of GeoGridPoints ─
    97→interface GridCell {
    98→  lat: number;
    99→  lng: number;
   100→  rank: number;
   101→  row: number; // 0 = north (top)
   102→  col: number; // 0 = west (left)
   103→}
   104→
   105→function buildGrid(points: GeoGridPoint[]): GridCell[] {
   106→  if (!points || points.length === 0) return [];
   107→  const lats = Array.from(new Set(points.map((p) => p.lat))).sort((a, b) => b - a); // desc (N→S)
   108→  const lngs = Array.from(new Set(points.map((p) => p.lng))).sort((a, b) => a - b); // asc (W→E)
   109→  return points.map((p) => ({
   110→    lat: p.lat,
   111→    lng: p.lng,
   112→    rank: p.rank,
   113→    row: lats.indexOf(p.lat),
   114→    col: lngs.indexOf(p.lng),
   115→  }));
   116→}

   416→// ── Geo-grid heatmap (centerpiece) ───────────────────────────────────
   417→function GeoGrid({ keyword, locationName }: { keyword: SeoKeyword; locationName: string }) {
   418→  const cells = useMemo(() => buildGrid(keyword.grid), [keyword.grid]);
   419→
   420→  if (cells.length === 0) {
   421→    return (
   422→      <div className="py-12 text-center text-sm text-muted-foreground">
   423→        No geo-grid data available for this keyword.
   424→      </div>
   425→    );
   426→  }
   427→
   428→  // Determine grid size (assume square)
   429→  const size = Math.max(
   430→    ...cells.map((c) => c.row + 1),
   431→    ...cells.map((c) => c.col + 1),
   432→  );
   433→
   434→  // Build 2D array [row][col]
   435→  const grid: (GridCell | null)[][] = Array.from({ length: size }, () =>
   436→    Array.from({ length: size }, () => null),
   437→  );
   438→  cells.forEach((c) => {
   439→    grid[c.row][c.col] = c;
   440→  });
   441→
   442→  // Compute summary stats
   443→  const ranked = cells.filter((c) => c.rank > 0);
   444→  const top3 = ranked.filter((c) => c.rank <= 3).length;
   445→  const top10 = ranked.filter((c) => c.rank <= 10).length;
   446→  const avg = ranked.length
   447→    ? Math.round((ranked.reduce((s, c) => s + c.rank, 0) / ranked.length) * 10) / 10
   448→    : 0;
   449→
   450→  return (
   451→    <div className="space-y-4">
   452→      {/* Title row */}
   453→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-2">
   454→        <div>
   455→          <div className="text-xs text-muted-foreground">
   456→            {locationName} · {keyword.city}
   457→          </div>
   458→          <div className="text-base font-semibold">
   459→            Geo-Grid Ranking — <span className="text-primary">{keyword.keyword}</span>
   460→          </div>
   461→        </div>
   462→        <div className="flex items-center gap-2 flex-wrap">
   463→          <MiniStat label="Avg" value={avg ? `#${avg}` : "—"} />
   464→          <MiniStat label="Top 3" value={`${top3}/${ranked.length || cells.length}`} />
   465→          <MiniStat label="Top 10" value={`${top10}/${ranked.length || cells.length}`} />
   466→        </div>
   467→      </div>
   468→
   469→      {/* The grid */}
   470→      <div className="flex justify-center">
   471→        <div className="flex items-stretch gap-2">
   472→          {/* Y-axis label (N/S) */}
   473→          <div className="flex flex-col items-center justify-between py-1">
   474→            <span className="text-[10px] font-medium text-muted-foreground">N</span>
   475→            <Compass className="size-3 text-muted-foreground/60" />
   476→            <span className="text-[10px] font-medium text-muted-foreground">S</span>
   477→          </div>
   478→
   479→          <div>
   480→            {/* Top labels (W/E) */}
   481→            <div className="flex items-center justify-between px-1 mb-1">
   482→              <span className="text-[10px] font-medium text-muted-foreground">W</span>
   483→              <span className="text-[10px] text-muted-foreground/70">lat ↓ / lng →</span>
   484→              <span className="text-[10px] font-medium text-muted-foreground">E</span>
   485→            </div>
   486→
   487→            {/* The 5x5 grid */}
   488→            <div
   489→              className="grid gap-1.5"
   490→              style={{ gridTemplateColumns: `repeat(${size}, minmax(0, 1fr))` }}
   491→            >
   492→              {grid.flat().map((cell, idx) => {
   493→                if (!cell) {
   494→                  return (
   495→                    <div
   496→                      key={idx}
   497→                      className="size-12 sm:size-14 rounded-md bg-muted/40 border border-dashed"
   498→                    />
   499→                  );
   500→                }
   501→                const c = rankColor(cell.rank);
   502→                return (
   503→                  <div
   504→                    key={idx}
   505→                    title={`Rank #${cell.rank || "—"} · lat ${cell.lat.toFixed(4)}, lng ${cell.lng.toFixed(4)}`}
   506→                    className={cn(
   507→                      "size-12 sm:size-14 rounded-md flex items-center justify-center text-sm font-bold tabular-nums shadow-sm transition-transform hover:scale-105 cursor-default",
   508→                      c.bg, c.text,
   509→                    )}
   510→                  >
   511→                    {cell.rank > 0 ? cell.rank : "—"}
   512→                  </div>
   513→                );
   514→              })}
   515→            </div>
   516→          </div>
   517→        </div>
   518→      </div>
   519→
   520→      {/* Mobile legend */}
   521→      <div className="flex items-center justify-center gap-3 flex-wrap text-[11px] sm:hidden">
   522→        <LegendItem color="bg-emerald-500" label="1–3" />
   523→        <LegendItem color="bg-amber-500" label="4–10" />
   524→        <LegendItem color="bg-orange-500" label="11–20" />
   525→        <LegendItem color="bg-rose-500" label="21+" />
   526→        <LegendItem color="bg-slate-300 dark:bg-slate-700" label="N/A" />
   527→      </div>
   528→
   529→      {/* Helper note */}
   530→      <div className="flex items-start gap-2 text-[11px] text-muted-foreground bg-muted/30 rounded-md p-2.5">
   531→        <Info className="size-3.5 shrink-0 mt-0.5" />

   593→// ── Keyword table ────────────────────────────────────────────────────
   594→function KeywordTable({
   595→  keywords, isLoading, selectedKeywordId, onSelect,
   596→}: {
   597→  keywords: SeoKeyword[];
   598→  isLoading: boolean;
   599→  selectedKeywordId: string | null;
   600→  onSelect: (id: string) => void;
   601→}) {
   602→  const [sortKey, setSortKey] = useState<SortKey>("avgRank");
   603→  const [sortDir, setSortDir] = useState<SortDir>("asc");
   604→
   605→  const sorted = useMemo(() => {
   606→    const arr = [...keywords];
   607→    arr.sort((a, b) => {
   608→      let av: number | string = 0;
   609→      let bv: number | string = 0;
   610→      switch (sortKey) {
   611→        case "keyword": av = a.keyword.toLowerCase(); bv = b.keyword.toLowerCase(); break;
   612→        case "city": av = a.city.toLowerCase(); bv = b.city.toLowerCase(); break;
   613→        case "avgRank": av = a.avgRank || 99; bv = b.avgRank || 99; break;
   614→        case "topRank": av = a.topRank ?? 99; bv = b.topRank ?? 99; break;
   615→      }
   616→      if (av < bv) return sortDir === "asc" ? -1 : 1;
   617→      if (av > bv) return sortDir === "asc" ? 1 : -1;
   618→      return 0;
   619→    });
   620→    return arr;
   621→  }, [keywords, sortKey, sortDir]);
   622→
   623→  function toggleSort(key: SortKey) {
   624→    if (sortKey === key) {
   625→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   626→    } else {
   627→      setSortKey(key);
   628→      setSortDir(key === "keyword" || key === "city" ? "asc" : "asc");
   629→    }
   630→  }
   631→
   632→  if (isLoading) {
   633→    return <Skeleton className="h-64 w-full" />;
   634→  }
   635→
   636→  if (keywords.length === 0) {
   637→    return (
   638→      <div className="py-10 text-center text-sm text-muted-foreground">
   639→        No keywords tracked yet.
   640→      </div>
   641→    );
   642→  }
   643→
   644→  return (
   645→    <div className="max-h-[28rem] overflow-y-auto scroll-area -mx-1">
   646→      <Table>
   647→        <TableHeader>
   648→          <TableRow className="bg-muted/30">
   649→            <SortableHead label="Keyword" k="keyword" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} className="min-w-[180px]" />
   650→            <SortableHead label="City" k="city" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} />
   651→            <SortableHead label="Avg Rank" k="avgRank" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} align="right" />
   652→            <SortableHead label="Top Rank" k="topRank" sortKey={sortKey} sortDir={sortDir} onSort={toggleSort} align="right" />
   653→            <TableHead className="text-center">Grid Preview</TableHead>
   654→            <TableHead className="text-right">Trend</TableHead>
   655→          </TableRow>
   656→        </TableHeader>
   657→        <TableBody>
   658→          {sorted.map((k) => {
   659→            const isSelected = k.id === selectedKeywordId;
   660→            const trend = mockTrend(k.keyword);
   661→            return (
   662→              <TableRow
   663→                key={k.id}
   664→                onClick={() => onSelect(k.id)}
   665→                className={cn(
   666→                  "cursor-pointer",
   667→                  isSelected && "bg-primary/5",
   668→                )}
   669→              >
   670→                <TableCell className="font-medium">
   671→                  <div className="flex items-center gap-2">
   672→                    {isSelected && <span className="size-1.5 rounded-full bg-primary" />}
   673→                    <span className="truncate max-w-[220px]">{k.keyword}</span>
   674→                  </div>
   675→                </TableCell>
   676→                <TableCell>
   677→                  <Badge variant="outline" className="text-[10px]">{k.city}</Badge>
   678→                </TableCell>
   679→                <TableCell className="text-right">
   680→                  <span className={cn("font-bold tabular-nums", rankTextClass(k.avgRank))}>
   681→                    {k.avgRank ? `#${k.avgRank}` : "—"}
   682→                  </span>
   683→                </TableCell>
   684→                <TableCell className="text-right">
   685→                  {k.topRank ? (
   686→                    <span className={cn("font-semibold tabular-nums", rankTextClass(k.topRank))}>
   687→                      #{k.topRank}
   688→                    </span>
   689→                  ) : (
   690→                    <span className="text-muted-foreground">—</span>
   691→                  )}
   692→                </TableCell>
   693→                <TableCell className="text-center">
   694→                  <MiniGrid points={k.grid} />
   695→                </TableCell>
   696→                <TableCell className="text-right">
   697→                  <TrendBadge delta={trend.delta} dir={trend.dir} />
   698→                </TableCell>
   699→              </TableRow>
   700→            );
   701→          })}
   702→        </TableBody>
   703→      </Table>
   704→    </div>
   705→  );
   706→}

   741→function MiniGrid({ points }: { points: GeoGridPoint[] }) {
   742→  const cells = useMemo(() => buildGrid(points).slice(0, 25), [points]);
   743→  if (cells.length === 0) {
   744→    return <span className="text-xs text-muted-foreground">—</span>;
   745→  }
   746→  const size = Math.max(
   747→    ...cells.map((c) => c.row + 1),
   748→    ...cells.map((c) => c.col + 1),
   749→  );
   750→  const grid: (GridCell | null)[][] = Array.from({ length: size }, () =>
   751→    Array.from({ length: size }, () => null),
   752→  );
   753→  cells.forEach((c) => { grid[c.row][c.col] = c; });
   754→
   755→  return (
   756→    <div
   757→      className="inline-grid gap-[2px]"
   758→      style={{ gridTemplateColumns: `repeat(${size}, 1fr)` }}
   759→    >
   760→      {grid.flat().map((cell, idx) => (
   761→        <div
   762→          key={idx}
   763→          title={cell ? `Rank #${cell.rank || "—"}` : "No data"}
   764→          className={cn(
   765→            "size-2 rounded-[2px]",
   766→            cell ? rankColorSoft(cell.rank) : "bg-muted",
   767→          )}
   768→        />
   769→      ))}
   770→    </div>
   771→  );
   772→}

   774→function TrendBadge({ delta, dir }: { delta: number; dir: "up" | "down" | "flat" }) {
